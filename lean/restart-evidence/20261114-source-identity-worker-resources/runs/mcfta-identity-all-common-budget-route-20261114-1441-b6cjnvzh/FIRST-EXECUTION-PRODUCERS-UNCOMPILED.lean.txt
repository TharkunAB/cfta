-- A completed rejection advances caller/query state before the next row.

    (caller reached : Caller width) (before after : Cached) (queries saved : Queries)
    (outer : Core) (waiting : EnumerationWaitingIndex.Waiting)

    (observed : SourcePruningEnumerationSharedSelectedContinuations.ChildDispatch.run
      checkEmpty fuel fuel fuel fuel fuel scs incoming caller before queries =
      ((.ok (some raw), reached, saved, outer, waiting), some after))
    (continuable : raw = [] ∨ ∃ fragment, raw = [.ok (fragment, after.core)]) :
    SourcePruningEnumerationSharedSelectedContinuations.ChildDispatch.run checkEmpty
      (fuel + extra) (fuel + extra) (fuel + extra) (fuel + extra) (fuel + extra)
      scs incoming caller before queries =
      ((.ok (some raw), reached, saved, outer, waiting), some after)
