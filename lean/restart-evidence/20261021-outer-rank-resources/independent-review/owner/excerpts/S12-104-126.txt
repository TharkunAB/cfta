  104     (initial checked merge fourth).checked = checked ∧
  105     (initial checked merge fourth).merge = merge ∧ (initial checked merge fourth).fourth = fourth ∧
  106     (initial checked merge fourth).nodeResults = FlatMemoThunk.empty ∧
  107     (initial checked merge fourth).edgeResults = FlatMemoThunk.empty := ⟨rfl, rfl, rfl, rfl, rfl⟩

  110 theorem advance_tables (before : State width) (checked : Checked width) (fourth : Fourth) (merge : MergeMemo) :
  111     (advance before checked fourth merge).checked = checked ∧
  112     (advance before checked fourth merge).fourth = fourth ∧ (advance before checked fourth merge).merge = merge ∧
  113     (advance before checked fourth merge).nodeResults = before.nodeResults ∧
  114     (advance before checked fourth merge).edgeResults = before.edgeResults := ⟨rfl, rfl, rfl, rfl, rfl⟩
