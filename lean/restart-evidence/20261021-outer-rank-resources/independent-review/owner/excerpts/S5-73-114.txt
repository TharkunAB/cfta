   73 private theorem fold_cut (step : Out → Out) (n k : Nat) (bound : k ≤ n) (entry : Out) :
   74     fold step n entry = fold step (n-k) (fold step k entry) := by
   75   unfold fold
   76   conv =>
   77     lhs

   79 private theorem prefix_kept (step : Out → Out)
   80     (one : ∀ acc, acc.1 + 1 ≤ (step acc).1 ∧ Kept acc (step acc))
   81     (n k : Nat) (bound : k ≤ n) (entry : Out) : Kept (fold step k entry) (fold step n entry) := by
   82   rw [fold_cut step n k bound entry]
   83   exact (fold_bounds step one _ _).2
