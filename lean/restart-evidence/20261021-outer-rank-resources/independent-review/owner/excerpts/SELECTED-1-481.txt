   31 private noncomputable def images (budget count mass : Nat) (seed : Seed) : List Out :=
   32   (((List.range (count + 1)).flatMap fun a => (List.range (seed.2.mass + 1)).map fun b => (a,b)).filter
   33     (fun r => decide (r.1 < count ∨ (r.1 = count ∧ r.2 < mass)))).map

   35 private def fold (step : Out → Out) (n : Nat) (entry : Out) : Out :=
   36   (List.range n).foldl (fun acc _ => step acc) entry
   37 private def slot (all : Seed → List Out) (acc : Out) : Out :=
   38   let xs := all acc.2.2
   39   (acc.1 + 1 + maximum (xs.map fun x => x.1), max acc.2.1 (maximum (xs.map fun x => x.2.1)),
   40     (acc.2.2.1 ++ xs.flatMap (fun x => x.2.2.1), cover (xs.map fun x => x.2.2.2) acc.2.2.2))
   41 private noncomputable def row (budget : Nat) (all : Seed → List Out) (acc : Out) : Out :=

   48 private theorem equation (budget count mass : Nat) (seed : Seed) :
   49     (envelope budget count mass seed).2.2 = SourceCurrentOuterInitialSchedule.rebuild
   50       (fold (row budget (images budget count mass)) mass (0,0,seed)).2.2 := by
   51   have eta {R : (Nat × Nat) → Prop} (f : (Nat × Nat) → Seed → Out) :
   52       (fun child (_ : R child) => f child) = (fun child _ => f (child.1,child.2)) := by
   53     funext child smaller current
   54     cases child
   55     rfl
   56   conv =>
   57     lhs
   58     unfold envelope
   59     rw [WellFounded.fix_eq, eta]
   60   let ranks := fun (current : Seed) =>
   61     ((List.range (count+1)).flatMap fun a => (List.range (current.2.mass+1)).map fun b => (a,b)).filter
   62       (fun r => decide (r.1 < count ∨ (r.1 = count ∧ r.2 < mass)))
   63   change SourceCurrentOuterInitialSchedule.rebuild
   64     (fold (row budget (fun current => (ranks current).attach.map
   65       (fun child => envelope budget child.val.1 child.val.2 current))) mass (0,0,seed)).2.2 = _
   66   have images_eq (current : Seed) :
   67       (ranks current).attach.map (fun child => envelope budget child.val.1 child.val.2 current) = images budget count mass current :=
   68     List.attach_map_val (f := fun r : Nat × Nat => envelope budget r.1 r.2 current)
   69   simp only [images_eq]

   71 private def Kept (first last : Seed) : Prop := Le first.2 last.2 ∧ first.1 ⊆ last.1
   72 private theorem kept_trans {first middle last : Seed} (one : Kept first middle) (two : Kept middle last) :
   73     Kept first last := ⟨one.1.trans two.1, fun _ h => two.2 (one.2 h)⟩
   74 private theorem fold_kept (step : Out → Out) (one : ∀ acc, Kept acc.2.2 (step acc).2.2)
   75     (n : Nat) (entry : Out) : Kept entry.2.2 (fold step n entry).2.2 := by
   76   have general (values : List Nat) (acc : Out) : Kept acc.2.2 (values.foldl (fun x _ => step x) acc).2.2 := by
   77     induction values generalizing acc with
   78     | nil => exact ⟨Le.refl _, fun _ h => h⟩
   79     | cons head tail ih => exact kept_trans (one acc) (ih (step acc))
   80   exact general _ _
   81 private theorem fold_succ (step : Out → Out) (n : Nat) (entry : Out) :
   82     fold step (n+1) entry = step (fold step n entry) := by
   83   simp only [fold, List.range_succ, List.foldl_append, List.foldl_cons, List.foldl_nil]
   84 private theorem prefix_kept (step : Out → Out) (one : ∀ acc, Kept acc.2.2 (step acc).2.2)
   85     {n k : Nat} (bound : k ≤ n) (entry : Out) : Kept (fold step k entry).2.2 (fold step n entry).2.2 := by
   86   have cut : fold step n entry = fold step (n-k) (fold step k entry) := by
   87     unfold fold
   88     conv =>
   89       lhs
   90       rw [show n = k + (n-k) by omega, List.range_add, List.foldl_append, List.foldl_map]
   91   rw [cut]
   92   exact fold_kept step one _ _
   93 private theorem slot_kept (all : Seed → List Out) (acc : Out) : Kept acc.2.2 (slot all acc).2.2 :=
   94   ⟨cover_ge _ _, fun _ h => List.mem_append_left _ h⟩
   95 private theorem row_kept (budget : Nat) (all : Seed → List Out) (acc : Out) :
   96     Kept acc.2.2 (row budget all acc).2.2 := by
   97   dsimp only [row]
   98   generalize post_eq : SourceCurrentOuterInitialSchedule.parent budget acc.2.2 = post
   99   have parent := SourceCurrentOuterInitialSchedule.parent_ge budget acc.2.2
  100   rw [post_eq] at parent
  101   have walked : Kept post (fold (slot all) post.2.mass (0,0,post)).2.2 :=
  102     fold_kept (slot all) (slot_kept all) post.2.mass (0,0,post)
  103   exact kept_trans parent (kept_trans walked (SourceCurrentOuterInitialSchedule.rebuild_ge _))
  104 private theorem image_kept (all : Seed → List Out) (acc value : Out) (member : value ∈ all acc.2.2) :

  107   intro guard present

  110     (counts : a ≤ count) (masses : b ≤ seed.2.mass) (smaller : Prod.Lex Nat.lt Nat.lt (a,b) (count,mass)) :

  118 private theorem candidate_profile (budget count mass : Nat) (seed : Seed) (j : Nat) (rowBound : j < mass) :
  119     let all := images budget count mass
  120     let prior := fold (row budget all) j (0,0,seed)
  121     let post := SourceCurrentOuterInitialSchedule.parent budget prior.2.2
  122     ∀ k, k < post.2.mass →
  123       let current := (fold (slot all) k (0,0,post)).2.2
  124       ∀ a b, a ≤ count → b ≤ current.2.mass → Prod.Lex Nat.lt Nat.lt (a,b) (count,mass) →
  125         Le (envelope budget a b current).2.2.2 (envelope budget count mass seed).2.2.2 := by
  126   dsimp only
  127   let all := images budget count mass
  128   let prior := fold (row budget all) j (0,0,seed)
  129   generalize post_eq : SourceCurrentOuterInitialSchedule.parent budget prior.2.2 = post
  130   intro k slotBound a b counts masses smaller
  131   let current := fold (slot all) k (0,0,post)
  132   let walked := fold (slot all) post.2.mass (0,0,post)
  133   let completed := fold (row budget all) mass (0,0,seed)
  134   have member := image_member budget count mass current.2.2 a b counts masses smaller
  135   have selected := image_kept all current (envelope budget a b current.2.2) member
  136   have slots : Kept (slot all current).2.2 walked.2.2 := by
  137     have bound : k < post.2.mass := slotBound
  138     have suffix := prefix_kept (slot all) (slot_kept all) (n := post.2.mass) (k := k+1) (by omega) (0,0,post)
  139     rw [fold_succ (slot all) k (0,0,post)] at suffix
  140     exact suffix
  141   have rebuilt : Kept walked.2.2 (row budget all prior).2.2 := by
  142     dsimp only [row]
  143     rw [post_eq]
  144     exact SourceCurrentOuterInitialSchedule.rebuild_ge _
  145   have rows : Kept (row budget all prior).2.2 completed.2.2 := by
  146     have suffix := prefix_kept (row budget all) (row_kept budget all)
  147       (n := mass) (k := j+1) (by omega) (0,0,seed)
  148     rw [fold_succ (row budget all) j (0,0,seed)] at suffix
  149     exact suffix
  150   have final := SourceCurrentOuterInitialSchedule.rebuild_ge completed.2.2
  151   have last := kept_trans selected (kept_trans slots (kept_trans rebuilt (kept_trans rows final)))
  152   rw [equation budget count mass seed]
  153   exact last.1
