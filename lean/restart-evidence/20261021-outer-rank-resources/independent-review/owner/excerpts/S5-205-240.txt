  205 theorem stage_bounds (budget count mass : Nat) (seed : List Guard × Profile) :
  206     let images := fun (current : List Guard × Profile) =>
  207       (((List.range (count + 1)).flatMap fun a =>
  208         (List.range (current.2.mass + 1)).map fun b => (a,b)).filter
  209           (fun r => decide (r.1 < count ∨ (r.1 = count ∧ r.2 < mass)))).map
  210             (fun r => envelope budget r.1 r.2 current)
  211     let slotStep := fun (acc : Nat × Nat × (List Guard × Profile)) (_ : Nat) =>
  212       let xs := images acc.2.2
  213       (acc.1 + 1 + SourceCurrentInitialSelectionForecast.maximum (xs.map fun x => x.1),
  214         max acc.2.1 (SourceCurrentInitialSelectionForecast.maximum (xs.map fun x => x.2.1)),
  215         (acc.2.2.1 ++ xs.flatMap (fun x => x.2.2.1),
  216           SourceCurrentInitialSelectionForecast.cover (xs.map fun x => x.2.2.2) acc.2.2.2))
  217     let rowStep := fun (acc : Nat × Nat × (List Guard × Profile)) (_ : Nat) =>
  218       let post := SourceCurrentOuterInitialSchedule.parent budget acc.2.2
  219       let walked := (List.range post.2.mass).foldl slotStep (0,0,post)
  220       (acc.1 + 1 + walked.1,
  221         max acc.2.1 (max (SourceCurrentParentUniformEnvelope.fuel budget acc.2.2.1 acc.2.2.2) walked.2.1),
  222         SourceCurrentOuterInitialSchedule.rebuild walked.2.2)
  223     let total := envelope budget count mass seed
  224     ∀ rowIndex, rowIndex < mass →
  225       let prior := (List.range rowIndex).foldl rowStep (0,0,seed)
  226       let post := SourceCurrentOuterInitialSchedule.parent budget prior.2.2
  227       SourceCurrentParentUniformEnvelope.fuel budget prior.2.2.1 prior.2.2.2 ≤ total.2.1 ∧
  228       Le post.2 total.2.2.2 ∧ post.1 ⊆ total.2.2.1 ∧
  229       ∀ slotIndex, slotIndex ≤ post.2.mass →
  230         let current := (List.range slotIndex).foldl slotStep (0,0,post)
  231         prior.1 + current.1 + 2 ≤ total.1 ∧ current.2.1 ≤ total.2.1 ∧
  232         Le current.2.2.2 total.2.2.2 ∧ current.2.2.1 ⊆ total.2.2.1 ∧
  233         ∀ childCount childMass, slotIndex < post.2.mass → childCount ≤ count →
  234           childMass ≤ current.2.2.2.mass →
  235           Prod.Lex Nat.lt Nat.lt (childCount,childMass) (count,mass) →
  236           let child := envelope budget childCount childMass current.2.2
  237           child.1 + 3 + rowIndex + slotIndex ≤ total.1 ∧ child.2.1 ≤ total.2.1 ∧
  238           Le child.2.2.2 total.2.2.2 ∧ child.2.2.1 ⊆ total.2.2.1 := by
  239   rw [equation budget count mass seed]
  240   simpa only [outputs, ranks, fold, slot, row, finish] using finite_stages budget count mass seed
