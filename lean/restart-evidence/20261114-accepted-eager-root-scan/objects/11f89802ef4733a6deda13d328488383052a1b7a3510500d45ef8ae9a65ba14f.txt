from pathlib import Path
import datetime
import hashlib
import json
import os
import re
import shlex
import subprocess

root = Path('/home/tritlo/Code/mcfta')
runtime = Path(__file__).parent
sha = lambda value: hashlib.sha256(value).hexdigest()
group = next(x.split(':', 2)[2] for x in Path('/proc/self/cgroup').read_text().splitlines()
             if x.startswith('0::'))
base = Path('/sys/fs/cgroup') / group.lstrip('/')
limits = {name: (base / name).read_text().strip()
          for name in ['memory.max', 'memory.high', 'memory.swap.max']}
assert limits == {'memory.max': '4294967296', 'memory.high': '4294967296', 'memory.swap.max': '0'}

def snapshot():
    return '\n'.join(p.name + ' ' + sha(p.read_bytes())
                     for p in sorted((root / 'lean/MicroCFTA').glob('*.lean'))) + '\n'

before = snapshot()
(runtime / 'final-source-pins.txt').write_text(before)
assert len(before.splitlines()) == 5915
baseline = json.loads((runtime / 'baseline-live-source-pins.json').read_text())
for name, digest in baseline.items():
    if name != 'lean/MicroCFTA/Audit.lean':
        assert sha((root / name).read_bytes()) == digest, name
module = 'SourceSynthesisCurrentAcceptedEagerRootScanOrigin'
source = root / 'lean/MicroCFTA' / (module + '.lean')
assert sha(source.read_bytes()) == 'd857c9e3742a1773d3ab29fa91e3019f749009c2df0b80af42284cfb9576a900'
qualified = 'MicroCFTA.SourceSynthesisCurrentProgramEagerOutputs.accepted_initial_root_scan_origin'
environment = dict(os.environ)
environment.pop('LEAN_SYSROOT', None)
environment['LEAN_NUM_THREADS'] = '1'
checks = []
allowed = {'propext', 'Classical.choice', 'Quot.sound'}

def run(name, command, minimum_groups=0):
    result = subprocess.run(['nix-shell', str(root / 'shell.nix'), '--run', command],
                            cwd=root / 'lean', env=environment, capture_output=True, text=True)
    output = result.stdout + result.stderr
    (runtime / (name + '.txt')).write_text(output)
    groups = [set(x.strip() for x in group.split(',') if x.strip())
              for group in re.findall(r'depends on axioms:\s*\[([^]]*)\]', output, re.S)]
    nonstandard = sorted(set().union(*groups) - allowed) if groups else []
    item = {'name': name, 'exit': result.returncode, 'axiom_groups': len(groups),
            'nonstandard_axioms': nonstandard}
    checks.append(item)
    (runtime / 'final-partial-report.json').write_text(json.dumps(checks, indent=2) + '\n')
    print(json.dumps(item), flush=True)
    assert result.returncode == 0 and len(groups) >= minimum_groups and not nonstandard
    assert 'sorryAx' not in output

run('audit-build', 'exec lake build MicroCFTA.Audit')
strict = runtime / 'public-strict.lean'
strict.write_bytes(source.read_bytes() + ('\n#print axioms ' + qualified + '\n').encode())
run('public-strict', 'exec lake env lean -j1 -DwarningAsError=true ' + shlex.quote(str(strict)), 1)
imports = runtime / 'public-imports.lean'
imports.write_text('import MicroCFTA.' + module + '\n\n#check @' + qualified +
                   '\n#print axioms ' + qualified + '\n')
run('public-imports', 'exec lake env lean -j1 -DwarningAsError=true ' + shlex.quote(str(imports)), 1)
audit = runtime / 'full-Audit.lean'
audit.write_bytes((root / 'lean/MicroCFTA/Audit.lean').read_bytes())
run('audit-strict', 'exec lake env lean -j1 -DwarningAsError=true ' + shlex.quote(str(audit)), 1)
counts = re.findall(r'Checked axiom dependencies of (\d+) MicroCFTA theorems',
                    (runtime / 'audit-strict.txt').read_text())
assert len(counts) == 1 and int(counts[0]) > 95100
assert snapshot() == before
report = {'at': datetime.datetime.now().astimezone().isoformat(), 'limits': limits,
          'live_modules': 5915, 'audit_imports': sum(x.startswith('import ') for x in audit.read_text().splitlines()),
          'audited_theorems': int(counts[0]), 'source_tree_sha256': sha(before.encode()),
          'checks': checks, 'public_laws_added': 1,
          'old_live_modules_unchanged_except_Audit': len(baseline) - 1,
          'joint_completion_proved': False, 'default_Haskell_entry_correspondence_proved': False,
          'repository_haskell_changes': False}
(runtime / 'final-checked.json').write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps(report), flush=True)
