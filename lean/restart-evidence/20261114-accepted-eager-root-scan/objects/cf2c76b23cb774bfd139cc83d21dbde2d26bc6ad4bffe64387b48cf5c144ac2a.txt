sha = lambda b: hashlib.sha256(b).hexdigest()
manifest = {'proof_commit': 'bbb03345da4197750ebc14d27d9553ec46a1951c',
            'previous_checkpoint': '3f3777d157a4294eb2c11be07fb992d6731953cf',

def put(path, label, classification):
    data = path.read_bytes()
    digest = sha(data)
    target = packet / 'objects' / (digest + ('.patch' if path.suffix == '.patch' else '.txt'))
    target.write_bytes(data)
    item = {'label': label, 'object': str(target.relative_to(packet)), 'sha256': digest,
            'bytes': len(data), 'classification': classification}
    manifest['files'].append(item)
    return item

def external(path):
    item = {'path': str(path.relative_to(root)), 'sha256': sha(path.read_bytes())}
    if item not in manifest['external_sources']:
        manifest['external_sources'].append(item)
    return item

def delta(before, after, label):
    output = subprocess.run(['diff', '-U0', '--label', 'a/source', '--label', 'b/source',
                             str(before), str(after)], capture_output=True)
    assert output.returncode == 1
    patch = runtime / (label + '.patch')
    patch.write_bytes(output.stdout)
    item = put(patch, label + '.patch', 'exact GNU zero-context delta')
    manifest['deltas'].append({'label': label, 'before_sha256': sha(before.read_bytes()),
                               'after_sha256': sha(after.read_bytes()), 'patch': item['object']})
