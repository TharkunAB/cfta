from pathlib import Path
import datetime
import hashlib
import json
import os
import re
import shlex
import subprocess

root = Path('/home/tritlo/Code/mcfta')
runtime = Path(__file__).parent
pins = json.loads((runtime / 'source-pins.json').read_text())
group = next(line.split(':', 2)[2] for line in Path('/proc/self/cgroup').read_text().splitlines()
             if line.startswith('0::'))
base = Path('/sys/fs/cgroup') / group.lstrip('/')
limits = {name: (base / name).read_text().strip()
          for name in ['memory.max', 'memory.high', 'memory.swap.max']}
assert limits == {'memory.max': '4294967296', 'memory.high': '4294967296', 'memory.swap.max': '0'}
environment = dict(os.environ)
environment.pop('LEAN_SYSROOT', None)
environment['LEAN_NUM_THREADS'] = '1'
checks = []
modules = [Path(name).stem for name in pins]

def verify_pins():
    for name, digest in pins.items():
        assert hashlib.sha256((root / name).read_bytes()).hexdigest() == digest, name

def run(name, command, required_groups=0):
    result = subprocess.run(['nix-shell', str(root / 'shell.nix'), '--run', command],
                            cwd=root / 'lean', env=environment, capture_output=True, text=True)
    output = result.stdout + result.stderr
    (runtime / (name + '.txt')).write_text(output)
    groups = [set(x.strip() for x in group.split(',') if x.strip())
              for group in re.findall(r'depends on axioms:\s*\[([^]]*)\]', output, re.S)]
    bad = sorted(set().union(*groups) - {'propext', 'Classical.choice', 'Quot.sound'}) if groups else []
    item = {'name': name, 'exit': result.returncode, 'axiom_groups': len(groups), 'nonstandard_axioms': bad}
    checks.append(item)
    (runtime / 'checks.json').write_text(json.dumps(checks, indent=2) + '\n')
    print(json.dumps(item), flush=True)
    verify_pins()
    assert result.returncode == 0 and not bad and len(groups) >= required_groups
    assert 'sorryAx' not in output

verify_pins()
run('baseline-build', 'exec lake build ' + ' '.join('MicroCFTA.' + m for m in modules))
for module in modules:
    source = root / 'lean/MicroCFTA' / (module + '.lean')
    run('strict-' + module, 'exec lake env lean -j1 -DwarningAsError=true ' + shlex.quote(str(source)))
run('baseline-imports', 'exec lake env lean -j1 -DwarningAsError=true ' +
    shlex.quote(str(runtime / 'imports.lean')), 3)
report = {'at': datetime.datetime.now().astimezone().isoformat(), 'limits': limits,
          'classification': 'baseline before changing shared narrow backend',
          'checks': checks, 'source_pins_unchanged': True}
(runtime / 'checked.json').write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps(report), flush=True)
