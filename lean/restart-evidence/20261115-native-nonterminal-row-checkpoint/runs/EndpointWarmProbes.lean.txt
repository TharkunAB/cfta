# Initialized results for every actual nonterminal source body row

Original source facts supply the Mu, reference and ordered child clauses.
Exact node-shape equality aligns every source witness with the same rows.
Every actual row query retains its supplied fuel, plans and complete state.
-/

/-- Original source facts derive initialized exact row readback at every actual member. -/
theorem current_stored_universe_body_first_nonterminal_row_results
    [DecidableEq Symbol] [DecidableEq (EdgeKey Symbol Guard)] {width : Nat}
    (lib : Library) (query : Query) (signed : Int) (whole : Expression)
    (accepted : SourceSynthesisCurrentConstruction.firstOrderRaw lib query signed = .ok whole)
    (rootValue : Value Symbol Guard) (sourceHeap : InternedConstructorTemplates.State Symbol Guard)
    (built : InternedFreshOrdinaryConstruction.build (ordinarySymbols whole) [] empty =
      .done rootValue sourceHeap)
    (key : NodeKey) (entry : Interned NodeKey) (body depth ordinary : Value Symbol Guard)
    (stored : sourceHeap.nodeRecords key = some (.mu entry body depth ordinary))
    (machine : SourceRecursiveIntersectionKernel.Machine Symbol Guard)
    (machineValid : SourceRecursiveIntersectionKernel.Valid machine)
    (facts : Facts machine.heap) (growth : Grows sourceHeap machine.heap)
    (domains : Cache.State IntersectionFlatMemoKeyDomain.DomainKey) (cacheValid : Cache.Valid domains)
    (pattern : Value Symbol Guard) (_patternAvailable : Available machine.heap pattern)
    (_patternClosed : pattern.free = []) (patternId : Nat)
    (_patternIdentity : identity pattern = some patternId)
    (_patternKind : tag pattern = 1 ∨ tag pattern = 3) :
    ∃ (bodyEntry : Interned NodeKey) (bodyRows : List (EdgeValue Symbol Guard)),
      body = .node bodyEntry bodyRows 0 [.interned entry.identity] ∧
      let universeValue : Value Symbol Guard := .mu entry body depth ordinary
      let referenceValue : Value Symbol Guard := .reference (.interned entry.identity)
      let original := SourceRecursiveIntersectionEnvironment.enter domains IntersectionEmptyDomain.emptyKey
        universeValue pattern entry.identity patternId
      let bodyFirst := SourceRecursiveIntersectionEnvironment.enter original.2 original.1.key
        body universeValue bodyEntry.identity entry.identity
      let muTarget : NodeRequest Symbol Guard := ⟨bodyFirst.1, universeValue, referenceValue⟩
      let referenceTarget : NodeRequest Symbol Guard := ⟨bodyFirst.1, referenceValue, referenceValue⟩
      let cutoff : Value Symbol Guard := repeatedValue (pair bodyEntry.identity entry.identity)
      ∀ row ∈ bodyRows,
      ∀ (operations : SourceEqualityConstraintMemoAllocation.State)
        (_operationsValid : SourceEqualityConstraintMemoAllocation.Valid operations) (fuel : Nat),
      let start : SourceEqualityNativeDemand.State width :=
        ⟨SourceRecursiveIntersectionThunks.initial width machine, operations⟩
      InputsThrough fuel (.code (.edge ⟨bodyFirst.1, row, row⟩ .pure)) start →
      FieldsThrough fuel (.code (.edge ⟨bodyFirst.1, row, row⟩ .pure)) start →
      Returned (NodeResult muTarget referenceTarget cutoff)
        (execute fuel (.code (.edge ⟨bodyFirst.1, row, row⟩ .pure)) start).state.native.nodes ∧
      Returned (EdgeResult muTarget referenceTarget cutoff)
        (execute fuel (.code (.edge ⟨bodyFirst.1, row, row⟩ .pure)) start).state.native.edges ∧
      ∀ (count : Nat) (result : EdgeValue Symbol Guard) (after : SourceEqualityNativeDemand.State width),
        count ≤ fuel → Steps count (.code (.edge ⟨bodyFirst.1, row, row⟩ .pure)) start (.code (.pure result)) after →
        result.children = row.children.map (input (replacement (.interned entry.identity) cutoff)) ∧
        result.entry.key = ⟨row.entry.key.symbol,
          (row.children.map (input (replacement (.interned entry.identity) cutoff))).map Value.referenceOf,
          SourceEqualityConstraintConstructors.top⟩ := by
  obtain ⟨bodyEntry, bodyRows, bodyShape, _bodyApart, sourceFacts⟩ :=
    SourceCurrentAllocatedNativeSourceBodyFirstActualMuCutoff.current_stored_universe_body_first_actual_mu_cutoff
      (width := width) lib query signed whole accepted rootValue sourceHeap built key entry body depth ordinary
      stored machine.heap facts growth domains cacheValid pattern _patternAvailable _patternClosed
      patternId _patternIdentity _patternKind
  obtain ⟨readEntry, readRows, readShape, readFacts⟩ :=
    SourceCurrentAllocatedNativeSourceBodyFirstReferenceRead.current_stored_universe_body_first_reference_read
      (width := width) lib query signed whole accepted rootValue sourceHeap built key entry body depth ordinary
      stored machine.heap facts growth domains cacheValid pattern _patternAvailable _patternClosed
      patternId _patternIdentity _patternKind
  have sameEntry : readEntry = bodyEntry := (Value.node.inj (readShape.symm.trans bodyShape)).1
  have sameReadRows : readRows = bodyRows := (Value.node.inj (readShape.symm.trans bodyShape)).2.1
  subst readEntry
  subst readRows
  obtain ⟨inputEntry, inputRows, inputShape, rowInputs⟩ :=
    SourceCurrentAllocatedNativeSourceBodyFirstNonterminalRowInputs.current_stored_universe_body_first_nonterminal_row_inputs
      (width := width) lib query signed whole accepted rootValue sourceHeap built key entry body depth ordinary
      stored machine.heap facts growth domains cacheValid pattern _patternAvailable _patternClosed
      patternId _patternIdentity _patternKind
  have sameInputEntry : inputEntry = bodyEntry := (Value.node.inj (inputShape.symm.trans bodyShape)).1
  have sameInputRows : inputRows = bodyRows := (Value.node.inj (inputShape.symm.trans bodyShape)).2.1
  subst inputEntry
  subst inputRows
  obtain ⟨classifiedEntry, _marker, arrow, rows, classifiedShape, _bodyAvailable, _markerShape,
      _arrowChildren, _arrowKey, _catalog, _markerClosed, _markerAvailable, _markerAtoms,
      childKinds, _arrowPresent⟩ :=
    SourceSynthesisCurrentUniverseBodyChildClassification.current_stored_universe_body_physical_children_after_growth
      lib query signed whole accepted rootValue sourceHeap built key entry body depth ordinary stored
      machine.heap facts growth
  have sameClassifiedEntry : classifiedEntry = bodyEntry :=
    (Value.node.inj (classifiedShape.symm.trans bodyShape)).1
  have sameClassifiedRows : bodyRows = canonicalEdges (arrow :: rows) :=
    (Value.node.inj (bodyShape.symm.trans classifiedShape)).2.1
  subst classifiedEntry
  let universeValue : Value Symbol Guard := .mu entry body depth ordinary
  let referenceValue : Value Symbol Guard := .reference (.interned entry.identity)
  let original := SourceRecursiveIntersectionEnvironment.enter domains IntersectionEmptyDomain.emptyKey
    universeValue pattern entry.identity patternId
  let bodyFirst := SourceRecursiveIntersectionEnvironment.enter original.2 original.1.key
    body universeValue bodyEntry.identity entry.identity
  let muTarget : NodeRequest Symbol Guard := ⟨bodyFirst.1, universeValue, referenceValue⟩
  let referenceTarget : NodeRequest Symbol Guard := ⟨bodyFirst.1, referenceValue, referenceValue⟩
  let cutoff : Value Symbol Guard := repeatedValue (pair bodyEntry.identity entry.identity)
  have selected : decision muTarget.domain.key muTarget.left muTarget.right = .leftMu := sourceFacts.1
  have leftIdentity : identity muTarget.left = some entry.identity := rfl
  have rightIdentity : identity muTarget.right = some entry.identity := rfl
  have childPure : ∀ current : Cache.State IntersectionFlatMemoKeyDomain.DomainKey, Cache.Valid current →
      compileNode ⟨(SourceRecursiveIntersectionEnvironment.enter current muTarget.domain.key
        muTarget.left muTarget.right entry.identity entry.identity).1,
        SourceRecursiveIntersectionKernel.body muTarget.left,
        SourceRecursiveIntersectionKernel.body muTarget.right⟩ = .pure cutoff := by
    intro current valid
    exact (sourceFacts.2.2.2.1 current valid).2.2
  have absent : let binder := pair entry.identity entry.identity
      Reference.intersection binder.1 binder.2 ∉ cutoff.free := sourceFacts.2.2.1
  have read : ∀ current : SourceEqualityNativeDemand.State width,
      SourceCurrentAllocatedNativePhysicalCache.Valid current.native →
      Grows sourceHeap current.native.machine.heap →
      SourceEqualityNativeDemand.step (.code (compileNode referenceTarget)) current =
        .yielded (.code (requestNode muTarget.domain muTarget.left muTarget.right)) current := by
    intro current physical sourceGrowth
    exact (readFacts.2.2.2 current physical.machine.1 sourceGrowth).2
  refine ⟨bodyEntry, bodyRows, bodyShape, ?_⟩
  dsimp only
  intro row member operations operationsValid fuel plan fields
  have rowTop : row.entry.key.constraint = SourceEqualityConstraintConstructors.top :=
    (childKinds row (by rw [← sameClassifiedRows]; exact member)).2.1
  have completed := SourceCurrentAllocatedNativeReferenceTopRowQueryGlobalLaws.initial_edge_query_returned_and_raw_results
    muTarget referenceTarget cutoff entry.identity entry.identity selected leftIdentity rightIdentity childPure absent
    sourceHeap read machine machineValid growth operations operationsValid fuel ⟨bodyFirst.1, row, row⟩ plan fields
  refine ⟨completed.1.1, completed.1.2, ?_⟩
  intro count result after bounded actual
  exact (completed.2 count result after bounded actual).2 rfl rowTop
    (row.children.map (input (replacement (.interned entry.identity) cutoff))) (rowInputs row member)

end MicroCFTA.SourceCurrentAllocatedNativeBodyFirstNonterminalRowIntegrationProbe
