local notation "Constraint" => SourceSymbolicIndex.Constraint FullSyntax.Formula
local notation "Cached" => EnumerationFlatCached.Cached Symbol Constraint
local notation "Core" => EnumerationFlatState.State Symbol Constraint
local notation "Caller" => SourceCurrentAllocatedPartialState.State
local notation "Queries" => SourceEndpointMemo.Memos
variable {Typing : Type} {width : Nat} {Output : Type u} {Failure : Type v}
local notation "Result" => SourcePruningEnumerationNativeContinuations.HookResult Output Failure width

/-- Run the fragment callback once before its actual recursive continuation. -/
def continueWith
    (callbackFailure : Nat → List (Except Error ((Bool × Metadata Typing) × Core)) → Failure)
    (recur : Metadata Typing → Caller width → Queries → Cached → Option Next → Result)
    (graph : Graph Symbol Constraint) (templates : List (Template.Pattern Symbol))
    (stop : Option (Metadata Typing → Nat → Bool))
    (order : Metadata Typing → List Nat → Option Nat)
    (callbackFuel : Nat) (metadata : Metadata Typing) (uv : Nat)
    (fragment : Fragments.Fragment Symbol) (caller : Caller width) (queries : Queries)
    (before : Cached) : Result :=
  let raw := SourceFlatPruningOracle.recordTypes
    graph templates callbackFuel metadata uv (.inl fragment) before.core
  match raw with
  | [.ok ((true, _updated), _observed)] => .complete [] caller queries
  | [.ok ((false, updated), observed)] =>
      let after := EnumerationFlatCached.frame observed before
      recur updated caller queries after (SourceFlatPruningRestart.decision stop order updated after)
  | _ => .stopped [] (callbackFailure uv raw) (some (caller, queries))

/-- An actual singleton callback determines the exact next cache and decision. -/
theorem continueWith_exact
    (callbackFailure : Nat → List (Except Error ((Bool × Metadata Typing) × Core)) → Failure)
    (recur : Metadata Typing → Caller width → Queries → Cached → Option Next → Result)
    (graph : Graph Symbol Constraint) (templates : List (Template.Pattern Symbol))
    (stop : Option (Metadata Typing → Nat → Bool))
    (order : Metadata Typing → List Nat → Option Nat)
    (callbackFuel : Nat) (metadata : Metadata Typing) (uv : Nat)
    (fragment : Fragments.Fragment Symbol) (caller : Caller width) (queries : Queries)
    (before : Cached) (result : Bool × Metadata Typing) (after : Core)
    (observed : SourceFlatPruningOracle.recordTypes graph templates callbackFuel metadata uv
      (.inl fragment) before.core = [.ok (result, after)]) :
    continueWith callbackFailure recur graph templates stop order callbackFuel metadata uv fragment caller queries before =
      if result.1 then .complete [] caller queries
      else recur result.2 caller queries (EnumerationFlatCached.frame after before)
        (SourceFlatPruningRestart.decision stop order result.2 (EnumerationFlatCached.frame after before)) := by
  unfold continueWith
  rw [observed]
  rcases result with ⟨pruned, updated⟩
  cases pruned <;> rfl

end MicroCFTA.SourceFlatPruningContinuation
