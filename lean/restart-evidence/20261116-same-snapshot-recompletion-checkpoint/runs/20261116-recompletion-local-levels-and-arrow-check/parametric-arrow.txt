  (bucket : List MicroCFTA.EnumerationState.Suspended)
  (rho : Nat → MicroCFTA.Language.Term MicroCFTA.SourcePruningTemplates.Symbol),
  MicroCFTA.EnumerationSchedule.IsRoot forest owner →
    MicroCFTA.UnionFind.Satisfies forest rho →
      MicroCFTA.EnumerationCorrectness.Avoids forest owner bucket →
        (∀ (sc : MicroCFTA.EnumerationState.Suspended),
            sc ∈ bucket →
              ∀ (path : MicroCFTA.EqualityTrie.Path),
                path ∈ MicroCFTA.EqualityTrie.fromPathTrie sc.trie → path = [1] ∨ path = [2]) →
          (∀ (sc : MicroCFTA.EnumerationState.Suspended),
              sc ∈ bucket →
                ∀ (position : Nat),

                    MicroCFTA.SourceSynthesisFiniteTypeWitnesses.TypeIn catalog (rho sc.var)) →
            (∀ (firstIndex secondIndex : Nat) (first second : MicroCFTA.EnumerationState.Suspended) (position : Nat),

                  bucket[secondIndex]? = some second →

                      [position] ∈ MicroCFTA.EqualityTrie.fromPathTrie second.trie → rho first.var = rho second.var) →
              ∃ argument result updated,
                MicroCFTA.SourceSynthesisFiniteTypeWitnesses.TypeIn catalog argument ∧

                                  sc ∈ bucket → MicroCFTA.EnumerationState.Suspended.Holds updated (updated owner) sc) ∧

                                  bucket[occurrence]? = some sc →
