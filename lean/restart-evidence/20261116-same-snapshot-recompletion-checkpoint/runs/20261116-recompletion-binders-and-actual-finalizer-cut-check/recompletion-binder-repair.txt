        (∀ (slot : Nat) (fragment : MicroCFTA.Fragments.Fragment Symbol),
            after.values[slot]? = some (MicroCFTA.EnumerationState.UVarValue.enumerated fragment) →
              before.values[slot]? = some (MicroCFTA.EnumerationState.UVarValue.enumerated fragment) ∨
                MicroCFTA.EnumerationDependencies.FragmentAvoid after.representatives held fragment) →
          (∀ (slot : Nat) (fragment : MicroCFTA.Fragments.Fragment Symbol),

                level slot < cut) →

@MicroCFTA.EnumerationPendingAssignment.recomplete_with_base_on : ∀ {Symbol Guard : Type}
  (state : MicroCFTA.EnumerationState.State Symbol Guard),
  MicroCFTA.EnumerationSchedule.Safe state →
    ∀ (roots : Nat → Nat),
      MicroCFTA.EnumerationSchedule.Roots state.representatives roots →
        ∀ (level : Nat → Nat) (frontier : Nat),
          MicroCFTA.EnumerationOrder.Stratified state level frontier →
            ∀ (current base : Nat → MicroCFTA.Language.Term Symbol),
              MicroCFTA.UnionFind.Satisfies state.representatives current →
                (∀ (slot : Nat) (fragment : MicroCFTA.Fragments.Fragment Symbol),
                    state.values[slot]? = some (MicroCFTA.EnumerationState.UVarValue.enumerated fragment) →
                      MicroCFTA.Fragments.fill current fragment = current slot) →
                  MicroCFTA.UnionFind.Satisfies state.representatives base →
                    ∀ (kept : Nat → Prop),
                      (∀ (uv : Nat),
                          kept uv →
                            ∀ (fragment : MicroCFTA.Fragments.Fragment Symbol),
                              state.values[roots uv]? =
                                  some (MicroCFTA.EnumerationState.UVarValue.enumerated fragment) →
                                ∀ (hole : Nat), MicroCFTA.EnumerationDependencies.HasHole fragment hole → kept hole) →
                        (∀ (uv : Nat), kept uv → base uv = current uv) →
                          have updated :=
                            MicroCFTA.EnumerationPendingAssignment.evaluateSlotsWith state.values roots base
                              (frontier + 1);
                          MicroCFTA.UnionFind.Satisfies state.representatives updated ∧
                            (∀ (uv : Nat) (fragment : MicroCFTA.Fragments.Fragment Symbol),
                                state.values[uv]? = some (MicroCFTA.EnumerationState.UVarValue.enumerated fragment) →
                                  updated uv = MicroCFTA.Fragments.fill updated fragment) ∧
                              (∀ (uv : Nat) (node : Option Nat) (bucket : List MicroCFTA.EnumerationState.Suspended),
                                  state.values[uv]? =
                                      some (MicroCFTA.EnumerationState.UVarValue.unenumerated node bucket) →
                                    updated uv = base uv) ∧
                                ∀ (uv : Nat), kept uv → updated uv = current uv
