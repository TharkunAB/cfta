   43 theorem edge_no_memo_failure (fuel g s r e p h m c n budget : Nat) (request : EdgeRequest)
   44     (before : Caller width) (valid : Valid before) (idle : EdgeIdle before)
   45     (handle : Nat) (observation : FlatMemoThunk.Observation EdgeRequest Edge) (after : Caller width) :
   46     SourceCurrentAllocatedPartialWorker.edgeCall fuel g s r e p h m c n budget request before ≠
   47       .failed (.edgeMemo handle observation) after := by
   48   intro failed
   49   rcases started : beginEdge request before with ⟨selected,phase,running⟩
   50   cases phase with
   51   | ready value =>
   52     simp only [SourceCurrentAllocatedPartialWorker.edgeCall,started] at failed
   53     cases failed
   54   | evaluate original =>
   55     simp only [SourceCurrentAllocatedPartialWorker.edgeCall,started,
   56       SourceCurrentAllocatedPartialWorker.edgeBody] at failed
   57     split at failed <;> simp only [SourceCurrentAllocatedPartialWorker.Outcome.bind] at failed <;> cases failed
   58   | busy =>
   59     have safe := SourceCurrentOuterEdgeIdle.edge_call fuel g s r e p h m c n budget request before idle
   60     simp only [SourceCurrentAllocatedPartialWorker.edgeCall,started] at failed
   61     cases failed
   62     simp only [SourceCurrentAllocatedPartialWorker.edgeCall,started,SourceCurrentOuterEdgeIdle.Safe,
   63       SourceCurrentOuterEdgeIdle.NoEdgeBusy] at safe
   64   | absent =>
   65     exact not_absent request before valid (congrArg (fun result => result.2.1) started)
