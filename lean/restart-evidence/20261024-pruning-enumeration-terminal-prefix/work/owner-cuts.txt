/-- A successful branch of an actual state action. Errors remain separate list entries. -/
def Success (action : EnumerateM Symbol Guard α) (before : State Symbol Guard)
    (value : α) (after : State Symbol Guard) : Prop :=
  Except.ok (value, after) ∈ ExceptT.run (action before)

/-- Binding runs each successful branch in its own stored state and retains error branches. -/
theorem run_bind (action : EnumerateM Symbol Guard α) (next : α → EnumerateM Symbol Guard β)
    (state : State Symbol Guard) :
    ExceptT.run ((action >>= next) state) = (ExceptT.run (action state)).flatMap
      (fun result => match result with
        | .error error => [.error error]
        | .ok (value, after) => ExceptT.run (next value after)) := by
  simp only [bind, StateT.bind, ExceptT.run, ExceptT.bind, ExceptT.mk]
  congr 1
  funext result
  cases result with
  | error error => rfl
  | ok pair => cases pair; rfl

/-- Successful sequential execution is exactly composition through one successful intermediate state. -/
theorem success_bind (action : EnumerateM Symbol Guard α) (next : α → EnumerateM Symbol Guard β)
    (before : State Symbol Guard) (value : β) (after : State Symbol Guard) :
    Success (action >>= next) before value after ↔
      ∃ middle intermediate, Success action before middle intermediate ∧ Success (next middle) intermediate value after := by
  simp only [Success, run_bind, List.mem_flatMap]
  constructor
  · rintro ⟨result, member, continued⟩
    cases result with
    | error error => simp at continued
    | ok pair => exact ⟨pair.1, pair.2, member, continued⟩
  · rintro ⟨middle, intermediate, member, continued⟩
    exact ⟨.ok (middle, intermediate), member, continued⟩

/-- A deterministic first step supplies exactly its returned state to the next step. -/
theorem bind_single (action : EnumerateM Symbol Guard α) (next : α → EnumerateM Symbol Guard β)
    (before after : State Symbol Guard) (value : α)
    (result : action before = [.ok (value, after)]) :
    (action >>= next) before = next value after := by
  change ExceptT.run ((action >>= next) before) = ExceptT.run (next value after)
  rw [run_bind]
  change (action before : List (Except Error (α × State Symbol Guard))).flatMap _ = _
  rw [result]
  simp

def Retains (first second : EnumerateM Symbol Guard α) : Prop :=
  ∀ before value after, Success first before value after → Success second before value after

/-- Sequential actions compose successful-branch inclusion. -/
theorem retains_bind (first second : EnumerateM Symbol Guard α)
    (left right : α → EnumerateM Symbol Guard β) (initial : Retains first second)
    (continued : ∀ value, Retains (left value) (right value)) :
    Retains (first >>= left) (second >>= right) := by
  intro before result after success
  obtain ⟨value, middle, step, rest⟩ := (success_bind _ _ _ _ _).mp success
  exact (success_bind _ _ _ _ _).mpr ⟨value, middle, initial _ _ _ step, continued value _ _ _ rest⟩

/-- An unchanged continuation retains successful branches. -/
theorem retains_refl (action : EnumerateM Symbol Guard α) : Retains action action := fun _ _ _ success => success

/-- List traversal retains each successful ordered result. -/
theorem retains_mapM (left right : α → EnumerateM Symbol Guard β)
    (step : ∀ input, Retains (left input) (right input)) (inputs : List α) :
    Retains (inputs.mapM left) (inputs.mapM right) := by
  induction inputs with
  | nil => exact retains_refl _
  | cons input rest ih =>
      simp only [List.mapM_cons]
      apply retains_bind _ _ _ _ (step input)
      intro output
      exact retains_bind _ _ _ _ ih (fun _ => retains_refl _)

/-- A fold retains its successful accumulator and state. -/
theorem retains_foldlM (left right : α → β → EnumerateM Symbol Guard α)
    (step : ∀ accumulator input, Retains (left accumulator input) (right accumulator input))
    (inputs : List β) (accumulator : α) :
    Retains (inputs.foldlM left accumulator) (inputs.foldlM right accumulator) := by
  induction inputs generalizing accumulator with
  | nil => exact retains_refl _
  | cons input rest ih =>
      simp only [List.foldlM_cons]
      exact retains_bind _ _ _ _ (step accumulator input) ih

/-- A discarded-result traversal retains every successful state. -/
theorem retains_forM (left right : α → EnumerateM Symbol Guard PUnit)
    (step : ∀ input, Retains (left input) (right input)) (inputs : List α) :
    Retains (inputs.forM left) (inputs.forM right) := by
  induction inputs with
  | nil => exact retains_refl _
  | cons input rest ih =>
      change Retains (left input >>= fun _ => rest.forM left) (right input >>= fun _ => rest.forM right)
      exact retains_bind _ _ _ _ (step input) (fun _ => ih)
