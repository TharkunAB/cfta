/-- Later memo state retains every original running request. -/
def Running (before after : State Key Request Value) : Prop :=
  ∀ handle request, before.cells handle = some ⟨request, .evaluating⟩ → after.cells handle = some ⟨request, .evaluating⟩

/-- Unchanged memo state retains all running cells. -/
theorem Running.refl (state : State Key Request Value) : Running state state := fun _ _ stored => stored

/-- Running-cell retention composes through consecutive actual states. -/
theorem Running.trans (first : Running before middle) (second : Running middle after) : Running before after :=
  fun handle request stored => second handle request (first handle request stored)
