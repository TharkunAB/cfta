def MicroCFTA.EnumerationFlatCached.frame : {Symbol Guard : Type} →
  MicroCFTA.EnumerationFlatState.State Symbol Guard →
    MicroCFTA.EnumerationFlatCached.Cached Symbol Guard → MicroCFTA.EnumerationFlatCached.Cached Symbol Guard :=
fun {Symbol Guard} core state => { core := core, expandable := state.expandable, waiting := state.waiting }
def MicroCFTA.EnumerationFlatCached.initial : {Symbol Guard : Type} →
  MicroCFTA.EnumerationState.Graph Symbol Guard → Nat → MicroCFTA.EnumerationFlatCached.Cached Symbol Guard :=
fun {Symbol Guard} graph root =>
  { core := MicroCFTA.EnumerationFlatState.initial root,
    expandable := fun uv =>
      if uv = 0 then
        MicroCFTA.EnumerationState.candidate graph (MicroCFTA.EnumerationState.UVarValue.unenumerated (some root) [])
      else false,
    waiting := fun x => none }
def MicroCFTA.EnumerationFlatNativeIndex.count : MicroCFTA.FlatUnionFind.Forest →
  Int →
    List MicroCFTA.EnumerationState.Suspended →
      MicroCFTA.EnumerationWaitingIndex.Waiting → MicroCFTA.EnumerationWaitingIndex.Waiting :=
fun forest change scs waiting =>
  MicroCFTA.EnumerationCanonicalWaiting.count (MicroCFTA.FlatUnionFind.root forest) change scs waiting
def MicroCFTA.EnumerationFlatNativeIndex.write : {EnumGuard NativeGuard : Type} →
  MicroCFTA.InternedConstructorTemplates.State MicroCFTA.SourcePruningTemplates.Symbol NativeGuard →
    Nat →
      MicroCFTA.EnumerationState.UVarValue MicroCFTA.SourcePruningTemplates.Symbol →
        List MicroCFTA.EnumerationState.Suspended →
          List MicroCFTA.EnumerationState.Suspended →
            MicroCFTA.EnumerationFlatCached.Cached MicroCFTA.SourcePruningTemplates.Symbol EnumGuard →
              Option (MicroCFTA.EnumerationFlatCached.Cached MicroCFTA.SourcePruningTemplates.Symbol EnumGuard) :=
fun {EnumGuard NativeGuard} heap target value gained lost before =>
  Option.map
    (fun flag =>
      {
        core :=
          have __src := before.core;
          { counter := __src.counter, representatives := __src.representatives,
            values := before.core.values.set target value, obligations := __src.obligations,
            recursionAncestry := __src.recursionAncestry, expansionAncestry := __src.expansionAncestry },
        expandable := fun other => if other = target then flag else before.expandable other,
        waiting :=
          MicroCFTA.EnumerationFlatNativeIndex.count before.core.representatives 1 gained
            (MicroCFTA.EnumerationFlatNativeIndex.count before.core.representatives (-1) lost before.waiting) })
    (MicroCFTA.SourcePruningEnumerationNativeIndex.candidate heap value)
def MicroCFTA.EnumerationFlatNativeIndex.WritesRelated : {EnumGuard : Type} →
  Option (MicroCFTA.EnumerationFlatCached.Cached MicroCFTA.SourcePruningTemplates.Symbol EnumGuard) →
    Option (MicroCFTA.EnumerationExpandableIndex.Cached MicroCFTA.SourcePruningTemplates.Symbol EnumGuard) → Prop :=
fun {EnumGuard} flat earlier =>
  match flat, earlier with
  | none, none => True
  | some flat, some earlier => MicroCFTA.EnumerationFlatCached.Related flat earlier
  | x, x_1 => False
