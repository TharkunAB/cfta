                  simpa only [foldObserved, cachedAdapterObserved, rawHeadObserved, observedHead] using member

                      apply atHead
                      simpa only [foldObserved, cachedAdapterObserved, rawHeadObserved, observedHead, actionRead] using member
                  | cons first tail =>
