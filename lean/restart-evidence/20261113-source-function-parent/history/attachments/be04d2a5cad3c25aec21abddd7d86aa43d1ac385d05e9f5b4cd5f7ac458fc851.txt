    (formulas : SourceGuards.Formulas FullSyntax.Formula)
    (solver : SourceGuards.Entailment FullSyntax.Formula)
    (lib : Library) (query : Query) (signed : Int) (whole : Expression)
    (accepted : SourceSynthesisCurrentConstruction.firstOrderRaw lib query signed = .ok whole)
    (rootValue : Value Symbol Guard) (heap : State Symbol Guard)
    (built : build (ordinarySymbols whole) [] empty = .done rootValue heap)
    (original : SourceSynthesisOriginalTypeAnnotations.OriginalType lib
      (.tFun (.tVar "a") (.tVar "a")))
    (functionRows : List Row) (sourceIndex : Nat) (name : String)
    (identity : functionRows[sourceIndex]? = some (.mk name top
      [generalize (allTypes top (SourceSynthesisCurrentContext.constructorCatalog lib query))
        (SourceSynthesisPr210TypeReflection.encodeTy (.tFun (.tVar "a") (.tVar "a")))]))
    (argument : Expression)
    {parentExpressions : List (EdgeSyntax Symbol Guard)} {before after : State Symbol Guard}
    {parentOutputs : List (EdgeValue Symbol Guard)} {parentIndex : Nat}
    (parent : SourceCurrentStoredMuRetainedAlignment.RowCall parentExpressions []
      before parentOutputs after heap parentIndex (literal "app")
      (SourceSynthesisGeneralizeSyntax.eqGuard
        [[[1], [2, 0, 0]], [[3, 0], [2, 0, 1]], [[0], [2, 0, 2]]])
      [ordinarySymbols (allTypes top (SourceSynthesisCurrentContext.constructorCatalog lib query)),
        ordinarySymbols (SourceSynthesisTypeSyntax.arrowMarker top),
        ordinarySymbols (.node functionRows), ordinarySymbols argument])
    (classes : EqualityNormalize.Constraints)
    (allThree : classes.Perm
      [EqualityNormalize.canonicalClass [[1], [2, 0, 0]],
       EqualityNormalize.canonicalClass [[3, 0], [2, 0, 1]],
       EqualityNormalize.canonicalClass [[0], [2, 0, 2]]])
    (results : List (Value Symbol Guard)) (readBefore readAfter : State Symbol Guard)
    (memos saved : EndpointMemoGateTrace.Memos)
    (reads : SourceCurrentRequiredQueryMemos.RequireAllReads
      (SourceEqualitySelectedRequired.paths classes) parent.output.children
      results readBefore readAfter memos saved) :
    ∃ functionValue functionStart functionAfter rows nodeHeap,
      parent.output.children.length = 4 ∧
      parent.output.children[2]? = some functionValue ∧
      InternedFreshOrdinaryContext.Context functionStart [] ∧
      build (ordinarySymbols (.node functionRows)) [] functionStart =
        .done functionValue functionAfter ∧
      Grows functionStart heap ∧ Grows functionAfter heap ∧
      InternedFreshOrdinaryConstruction.buildEdges
        (SourceSynthesisUniverseCallerBoundary.ordinaryEdges functionRows) [] functionStart =
        .done rows nodeHeap ∧
      run (makeNode rows) nodeHeap = (functionValue, functionAfter) ∧
      ∃ call : SourceCurrentStoredMuRetainedAlignment.RowCall
          (SourceSynthesisUniverseCallerBoundary.ordinaryEdges functionRows) []
          functionStart rows nodeHeap heap sourceIndex (literal name) top
          [ordinarySymbols (generalize
            (allTypes top (SourceSynthesisCurrentContext.constructorCatalog lib query))
            (SourceSynthesisPr210TypeReflection.encodeTy (.tFun (.tVar "a") (.tVar "a"))))],
        ∃ canonicalIndex : Nat, ∃ typeValue,
          (InternedEdgeResidency.nodeEdges functionValue)[canonicalIndex]? = some call.output ∧
          call.output.children = [typeValue] ∧ Available heap typeValue ∧ typeValue.free = [] ∧
          SourceEqualityAnnotationLanguage.Annotated
            (SourceSynthesisNativeTypeTrace.Origin
              (SourceSynthesisCurrentSourceGuards.encodedLibrary lib)) typeValue ∧
          (∀ ambient : InternedGuardedLanguage.Env Symbol,
            InternedGuardedLanguage.meaning (SourceEqualityPCSGuard.holds formulas solver)
              ambient typeValue
              (.node (literal "->")
                [.node (literal "(->)") [], .node (literal "var1") [], .node (literal "var1") []])) ∧
          ∃ functionEntry functionEdges functionDepth functionFree resultType marker argumentValue,
            functionValue = .node functionEntry functionEdges functionDepth functionFree ∧
            functionEdges[canonicalIndex]? = some call.output ∧
            parent.output.children = [resultType, marker, functionValue, argumentValue] ∧
            ∃ resultTypeResult markerResult functionResult argumentResult
                afterResult afterMarker afterFunction resultMemos markerMemos functionMemos,
              results = [resultTypeResult, markerResult, functionResult, argumentResult] ∧
              (∃ actual : SourceEqualityRequiredGraph.RequireNode (.node [])
                  resultType readBefore resultTypeResult afterResult,
                SourceCurrentRequiredQueryMemos.NodeReads actual memos resultMemos) ∧
              (∃ actual : SourceEqualityRequiredGraph.RequireNode (.node [])
                  marker afterResult markerResult afterMarker,
                SourceCurrentRequiredQueryMemos.NodeReads actual resultMemos markerMemos) ∧
              (∃ actual : SourceEqualityRequiredGraph.RequireNode
                  (.node [(0, SourceEqualityNeeded.build [[0], [1], [2]])])
                  functionValue afterMarker functionResult afterFunction,
                SourceCurrentRequiredQueryMemos.NodeReads actual markerMemos functionMemos) ∧
              (∃ actual : SourceEqualityRequiredGraph.RequireNode
                  (SourceEqualityNeeded.build [[0]])
                  argumentValue afterFunction argumentResult readAfter,
                SourceCurrentRequiredQueryMemos.NodeReads actual functionMemos saved) ∧
              ∃ prefixResults typeResult rebuilt suffixResults
                  rowBefore typeAfter edgeAfter rowsAfter rowMemos typeMemos,
                (∃ earlier : SourceEqualityRequiredGraph.RequireRows
                    [(0, SourceEqualityNeeded.build [[0], [1], [2]])]
                    (functionEdges.take canonicalIndex) afterMarker prefixResults rowBefore,
                  SourceCurrentRequiredQueryMemos.RowsReads earlier markerMemos rowMemos) ∧
                (∃ actual : SourceEqualityRequiredGraph.RequireNode
                    (SourceEqualityNeeded.build [[0], [1], [2]])
                    typeValue rowBefore typeResult typeAfter,
                  SourceCurrentRequiredQueryMemos.NodeReads actual rowMemos typeMemos) ∧
                run (makeEdge call.output.entry.key.symbol call.output.entry.key.constraint [typeResult])
                  typeAfter = (rebuilt, edgeAfter) ∧
                (∃ later : SourceEqualityRequiredGraph.RequireRows
                    [(0, SourceEqualityNeeded.build [[0], [1], [2]])]
                    (functionEdges.drop (canonicalIndex + 1)) edgeAfter suffixResults rowsAfter,
                  SourceCurrentRequiredQueryMemos.RowsReads later typeMemos functionMemos) ∧
                run (makeNode (prefixResults ++ rebuilt :: suffixResults)) rowsAfter =
                  (functionResult, afterFunction)
