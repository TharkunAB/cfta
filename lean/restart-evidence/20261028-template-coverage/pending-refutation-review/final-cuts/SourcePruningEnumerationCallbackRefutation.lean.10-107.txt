/-- The complete callback resource certificate also retains original template refutation. -/
theorem callback_prefix_complete_resources_refutes
    (holeKey : Typing → Nat → Key) (fits : Key → Symbol → Bool) (admits : Key → Node → Bool)
    (checkEmpty : Bool) (edgesOf : Node → List (InternedConstructorTemplates.EdgeValue Symbol Constraint))
    (stop : Option (Metadata Typing → Nat → Bool)) (order : Metadata Typing → List Nat → Option Nat)
    (packet : Packet formulas solver inventory caller memos metadata before)
    (allocated : FragmentBound before.core.counter fragment)
    (profile : Profile) (property : Constraint → Prop)
    (bounded : Fits profile caller.checked) (known : Heap property caller.checked.native.machine.heap)
    (language : NodeLanguage Symbol) (meaning : Constraint → Language.Term Symbol → Prop)
    (assignment : Nat → Language.Term Symbol)
    (satisfied : Satisfies language meaning assignment before.core)
    (filled : Fragments.fill assignment fragment = assignment uv)
    (refuted : ∀ group ∈ metadata.pending, @SourcePruningResume.Refutes Symbol
        (@instDecidableEqProd String FullSyntax.Formula inferInstance FormulaExecutable.formulaDecidableEq) assignment group)
    (avoids : SourcePruningTemplates.Avoids templates (Fragments.fill assignment fragment))
    (encoded : ∀ program, SourcePruningTemplates.ProgramAt (Fragments.fill assignment fragment) program →
      SourcePruningTemplates.ProgramEncoding program) :
    ∃ fuel result after,
      ((Packet formulas solver inventory caller memos result.2 (EnumerationExpandableIndex.frame after before) ∧
      RenderStable before.core after ∧ Frame before.core after ∧
      result.2.types = updatedTypes metadata uv (.inl fragment) ∧
      result.2.openHoles = metadata.openHoles ∧ result.2.typing = metadata.typing ∧
      ∀ extra,
        Success (recordTypes graph templates (fuel + extra) metadata uv (.inl fragment)) before.core result after ∧
        ExceptT.run (recordTypes graph templates (fuel + extra) metadata uv (.inl fragment) before.core) =
          [.ok (result, after)] ∧
        (∀ recur : Metadata Typing → Caller width → Queries → Cached → Option Next → Result,
          continueWith recur graph templates stop order (fuel + extra) metadata uv fragment caller memos.query before =
            if result.1 then .complete [] caller memos.query
            else recur result.2 caller memos.query (EnumerationExpandableIndex.frame after before)
              (SourcePruningEnumerationRestart.decision stop order result.2
                (EnumerationExpandableIndex.frame after before))) ∧
        (result.1 = false → ∀ selected,
          SourcePruningEnumerationRestart.decision stop order result.2
              (EnumerationExpandableIndex.frame after before) = some (.expand selected) →
          SourcePruningEnumerationSelectedStep.PrefixValid holeKey fits admits graph templates checkEmpty edgesOf
            stop order result.2 formulas solver inventory (fuel + extra) selected caller
            (EnumerationExpandableIndex.frame after before) memos
            (SourcePruningEnumerationSelectedStep.runPrefix holeKey fits admits graph templates stop order result.2
              (fuel + extra) (fuel + extra) (fuel + extra) (fuel + extra) (fuel + extra) (fuel + extra)
              selected caller (EnumerationExpandableIndex.frame after before) memos.query))) ∧
      (∀ extra, result.1 = false → ∀ selected,
        SourcePruningEnumerationRestart.decision stop order result.2
            (EnumerationExpandableIndex.frame after before) = some (.expand selected) →
        let observed := SourcePruningEnumerationSelectedStep.runPrefix holeKey fits admits graph templates
          stop order result.2 (fuel + extra) (fuel + extra) (fuel + extra)
          (fuel + extra) (fuel + extra) (fuel + extra) selected caller
          (EnumerationExpandableIndex.frame after before) memos.query
        SourcePruningEnumerationSelectedStep.Resources profile property caller observed ∧
        match observed with
        | .parked _ (some reachedCache) _ =>
            Packet formulas solver inventory caller memos result.2 reachedCache
        | .narrowed _ _ reached queries _ (some reachedCache) _ _ =>
            ∃ final, Post formulas solver inventory caller reached final ∧ final.query = queries ∧
              Packet formulas solver inventory reached final result.2 reachedCache
        | _ => True)) ∧
      result.1 = false ∧
      ∀ group ∈ result.2.pending, @SourcePruningResume.Refutes Symbol
        (@instDecidableEqProd String FullSyntax.Formula inferInstance FormulaExecutable.formulaDecidableEq) assignment group := by
  obtain ⟨fuel, result, after, retained⟩ :=
    callback_prefix_complete_resources formulas solver inventory graph templates metadata uv fragment caller memos before
      holeKey fits admits checkEmpty edgesOf stop order packet allocated profile property bounded known
  have success := (retained.1.2.2.2.2.2.2 0).1
  have rejected := recordTypes_inl_refutes graph language meaning assignment templates (fuel + 0) uv metadata fragment
    before.core after result packet.safe.1.total satisfied filled refuted avoids encoded success
  exact ⟨fuel, result, after, retained, rejected⟩
