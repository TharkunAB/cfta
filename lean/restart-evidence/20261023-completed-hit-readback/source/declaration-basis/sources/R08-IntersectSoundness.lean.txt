import MicroCFTA.IntersectDomain

/-!
# Language soundness of the recursive intersection schedule

Input languages satisfy the original constructor, union, and binder
equations. The proof follows the actual bounded source schedule. Generated
binders use least-language semantics. A successful output cannot introduce
a finite term absent from either input language.
-/
namespace MicroCFTA.IntersectSchedule
open Language Binding BinderLanguage

/-- A nonempty input reads its language through its globally unique identity. -/
def inputLanguage (languages : Nat → Lang Symbol) : Input → Lang Symbol
  | .empty => fun _ => False
  | input => languages (identity input)

/-- Independent input equations describe only valid interned node tags. -/
structure InputMeaning (view : View Symbol) where
  valid : Input → Prop
  languages : Nat → Lang Symbol
  body_valid : ∀ name, valid (.mu name) → valid (view.body name)
  body_exact : ∀ name, valid (.mu name) → languages name = inputLanguage languages (view.body name)
  reference_valid : ∀ name, valid (.ref name) → valid (.mu name)
  row_valid : ∀ name, valid (.node name) → ∀ row ∈ view.rows name, ∀ child ∈ row.children, valid child
  row_exact : ∀ name, valid (.node name) → ∀ term, languages name term ↔
    ∃ row ∈ view.rows name, ∃ terms, term = .node row.label terms ∧
      All₂ (inputLanguage languages) row.children terms ∧ row.guard term = true
  union_valid : ∀ name, valid (.union name) → ∀ child ∈ view.members name, valid child
  union_exact : ∀ name, valid (.union name) → ∀ term, languages name term ↔
    ∃ child ∈ view.members name, inputLanguage languages child term
  copy_exact : ∀ input, valid input → closed view input = true → ∀ environment,
    meaning environment (view.copy input) = inputLanguage languages input

/-- The common language named by one canonical identity pair. -/
def pairLanguage (languages : Nat → Lang Symbol) (key : Pair) : Lang Symbol :=
  fun term => languages key.1 term ∧ languages key.2 term

/-- Canonical identity order preserves the common language of the two operands. -/
theorem pair_language (languages : Nat → Lang Symbol) (left right : Input)
    (leftNonempty : left ≠ .empty) (rightNonempty : right ≠ .empty) (term : Term Symbol) :
    pairLanguage languages (pair left right) term ↔ inputLanguage languages left term ∧ inputLanguage languages right term := by
  have leftEq : inputLanguage languages left = languages (identity left) := by cases left <;> simp_all [inputLanguage]
  have rightEq : inputLanguage languages right = languages (identity right) := by cases right <;> simp_all [inputLanguage]
  rw [leftEq, rightEq]
  unfold pairLanguage pair
  by_cases ordered : identity left ≤ identity right
  · simp [Nat.min_eq_left ordered, Nat.max_eq_right ordered]
  · have reversed : identity right ≤ identity left := by omega
    simp [Nat.min_eq_right reversed, Nat.max_eq_left reversed, and_comm]

/-- Only canonical pairs occur in a source domain built from the empty domain. -/
def Canonical (domain : Domain) : Prop := ∀ key ∈ domain.seen, key.1 ≤ key.2

/-- Every remembered pair denotes a sublanguage of its original input intersection. -/
def SoundEnvironment (languages : Nat → Lang Symbol) (domain : Domain) (environment : Env Symbol) : Prop :=
  ∀ key ∈ domain.seen, Included (environment (pairName key)) (pairLanguage languages key)

/-- The source domain stores each recursive binder under its own identity. -/
def Typed (domain : Domain) : Prop := ∀ name input, lookup domain name = some input → input = .mu name

/-- Typed binder insertion preserves every lookup identity. -/
theorem typed_extend (domain : Domain) (typed : Typed domain) (names : List Nat) (key : Pair) :
    Typed (extend domain (names.map (fun name => (name, Input.mu name))) key) := by
  intro name input stored
  rw [extend_lookup] at stored
  split at stored
  · exact (Option.some.inj stored).symm
  · exact typed name input stored

/-- Every pair generated by the source constructor is canonical. -/
theorem canonical_pair (left right : Input) : (pair left right).1 ≤ (pair left right).2 := by
  exact Nat.le_trans (Nat.min_le_left _ _) (Nat.le_max_left _ _)

/-- Adding a new canonical pair preserves the domain's canonical-name invariant. -/
theorem canonical_extend (domain : Domain) (canonical : Canonical domain)
    (bindings : List (Nat × Input)) (left right : Input) : Canonical (extend domain bindings (pair left right)) := by
  intro key member
  rcases List.mem_cons.mp member with rfl | member
  · exact canonical_pair left right
  · exact canonical key member

/-- Interpreting a fresh binder by its original common language preserves all enclosing meanings. -/
theorem sound_bind (languages : Nat → Lang Symbol) (domain : Domain) (environment : Env Symbol)
    (canonical : Canonical domain) (sound : SoundEnvironment languages domain environment)
    (bindings : List (Nat × Input)) (key : Pair) (ordered : key.1 ≤ key.2) (fresh : key ∉ domain.seen) :
    SoundEnvironment languages (extend domain bindings key)
      (BinderLanguage.bind environment (pairName key) (pairLanguage languages key)) := by
  intro found member term accepted
  rcases List.mem_cons.mp member with rfl | member
  · simpa [BinderLanguage.bind] using accepted
  · have different : pairName found ≠ pairName key := by
      intro same
      have equal := pairName_injective found key (canonical found member) ordered same
      exact fresh (equal ▸ member)
    exact sound found member term (by simpa [BinderLanguage.bind, different] using accepted)

/-- Least binding is sound when the original common language is closed under the constructed body. -/
theorem maybeMu_sound (languages : Nat → Lang Symbol) (domain : Domain) (environment : Env Symbol)
    (canonical : Canonical domain) (sound : SoundEnvironment languages domain environment)
    (bindings : List (Nat × Input)) (key : Pair) (ordered : key.1 ≤ key.2) (fresh : key ∉ domain.seen)
    (body : Shape (Payload Symbol))
    (bodySound : ∀ inner, SoundEnvironment languages (extend domain bindings key) inner →
      Included (meaning inner body) (pairLanguage languages key)) :
    Included (meaning environment (maybeMu key body)) (pairLanguage languages key) := by
  rw [maybeMu_exact]
  rw [meaning]
  exact least_le _ _ (bodySound _ (sound_bind languages domain environment canonical sound bindings key ordered fresh))

/-- Paired child executions preserve both complete ordered input forests. -/
theorem forest_common (languages : Nat → Lang Symbol) (environment : Env Symbol)
    (left right : List Input) (shapes : List (Shape (Payload Symbol))) (sameLength : left.length = right.length)
    (paired : All₂ (fun inputs shape => Included (meaning environment shape)
      (fun term => inputLanguage languages inputs.1 term ∧ inputLanguage languages inputs.2 term)) (left.zip right) shapes)
    (terms : List (Term Symbol)) (accepted : forest environment shapes terms) :
    All₂ (inputLanguage languages) left terms ∧ All₂ (inputLanguage languages) right terms := by
  induction left generalizing right shapes terms with
  | nil =>
    have rightEmpty : right = [] := List.length_eq_zero_iff.mp sameLength.symm
    subst right
    cases paired
    cases terms with
    | nil => exact ⟨.nil, .nil⟩
    | cons _ _ => simp [forest] at accepted
  | cons first rest ih =>
    cases right with
    | nil => simp at sameLength
    | cons second others =>
      cases paired with
      | cons head tail =>
        cases terms with
        | nil => simp [forest] at accepted
        | cons term terms =>
          rw [forest] at accepted
          have common := head term accepted.1
          have remainder := ih others _ (by simpa using sameLength) tail terms accepted.2
          exact ⟨.cons common.1 remainder.1, .cons common.2 remainder.2⟩

/-- A constructed row preserves both child languages and both original complete guards. -/
def EdgeCommon (languages : Nat → Lang Symbol) (left right : Row Symbol) (term : Term Symbol) : Prop :=
  ∃ terms, term = .node left.label terms ∧ All₂ (inputLanguage languages) left.children terms ∧
    All₂ (inputLanguage languages) right.children terms ∧ left.guard term = true ∧ right.guard term = true

/-- Every successful recursive source intersection is a sublanguage of both independent input languages. -/
theorem run_sound [DecidableEq Symbol] (view : View Symbol) (model : InputMeaning view) (fuel : Nat) :
    (∀ domain left right result, Typed domain → Canonical domain → model.valid left → model.valid right →
      run view fuel domain left right = .ok result → ∀ environment,
      SoundEnvironment model.languages domain environment →
      Included (meaning environment result) (fun term => inputLanguage model.languages left term ∧ inputLanguage model.languages right term)) ∧
    (∀ domain left right result, Typed domain → Canonical domain →
      (∀ child ∈ left.children, model.valid child) → (∀ child ∈ right.children, model.valid child) →
      edge view fuel domain left right = .ok result → ∀ environment,
      SoundEnvironment model.languages domain environment →
      Included (meaning environment result) (EdgeCommon model.languages left right)) := by
  induction fuel with
  | zero => constructor <;> intro domain left right result typed canonical leftValid rightValid completed <;> cases completed
  | succ fuel ih =>
    constructor
    · intro domain left right result typed canonical leftValid rightValid completed environment environmentSound term accepted
      rw [run.eq_def] at completed
      dsimp only at completed
      by_cases empty : left = .empty ∨ right = .empty
      · rw [ite_eq_left empty] at completed
        cases completed
        simp [meaning] at accepted
      rw [ite_eq_right empty] at completed
      have leftNonempty : left ≠ .empty := fun same => empty (Or.inl same)
      have rightNonempty : right ≠ .empty := fun same => empty (Or.inr same)
      by_cases reset : (closed view left && closed view right && !domain.free.isEmpty) = true
      · rw [ite_eq_left reset] at completed
        exact ih.1 {} left right result (by intro name input stored; cases stored)
          (by intro key member; cases member) leftValid rightValid completed environment
          (by intro key member; cases member) term accepted
      rw [ite_eq_right reset] at completed
      by_cases same : left = right ∧ closed view left = true
      · rw [ite_eq_left same] at completed
        cases completed
        have member : inputLanguage model.languages left term := by
          rw [← model.copy_exact left leftValid same.2 environment]
          exact accepted
        exact ⟨member, same.1 ▸ member⟩
      rw [ite_eq_right same] at completed
      by_cases swap : greater left right = true
      · rw [ite_eq_left swap] at completed
        exact (ih.1 domain right left result typed canonical rightValid leftValid completed environment environmentSound term accepted).symm
      rw [ite_eq_right swap] at completed
      by_cases seen : pair left right ∈ domain.seen
      · rw [ite_eq_left seen] at completed
        cases completed
        rw [meaning] at accepted
        exact (pair_language model.languages left right leftNonempty rightNonempty term).mp
          (environmentSound _ seen term accepted)
      rw [ite_eq_right seen] at completed
      split at completed
      · next first second =>
        obtain ⟨body, built, rfl⟩ := map_success _ _ _ completed
        apply (pair_language model.languages _ _ leftNonempty rightNonempty term).mp
        refine maybeMu_sound model.languages domain environment canonical environmentSound
          [(first, .mu first), (second, .mu second)] _ (canonical_pair _ _) seen body ?_ term accepted
        intro inner innerSound output bodyAccepted
        have common := ih.1 _ _ _ body (typed_extend domain typed [first, second] _)
          (canonical_extend domain canonical _ _ _) (model.body_valid first leftValid) (model.body_valid second rightValid)
          built inner innerSound output bodyAccepted
        apply (pair_language model.languages (.mu first) (.mu second) (by intro h; cases h) (by intro h; cases h) output).mpr
        change model.languages first output ∧ model.languages second output
        rw [model.body_exact first leftValid, model.body_exact second rightValid]
        exact common
      · next first _ =>
        obtain ⟨body, built, rfl⟩ := map_success _ _ _ completed
        apply (pair_language model.languages _ _ leftNonempty rightNonempty term).mp
        refine maybeMu_sound model.languages domain environment canonical environmentSound
          [(first, .mu first)] _ (canonical_pair _ _) seen body ?_ term accepted
        intro inner innerSound output bodyAccepted
        have common := ih.1 _ _ _ body (typed_extend domain typed [first] _)
          (canonical_extend domain canonical _ _ _) (model.body_valid first leftValid) rightValid
          built inner innerSound output bodyAccepted
        apply (pair_language model.languages (.mu first) right (by intro h; cases h) rightNonempty output).mpr
        change model.languages first output ∧ inputLanguage model.languages right output
        rw [model.body_exact first leftValid]
        exact common
      · next second _ =>
        obtain ⟨body, built, rfl⟩ := map_success _ _ _ completed
        apply (pair_language model.languages _ _ leftNonempty rightNonempty term).mp
        refine maybeMu_sound model.languages domain environment canonical environmentSound
          [(second, .mu second)] _ (canonical_pair _ _) seen body ?_ term accepted
        intro inner innerSound output bodyAccepted
        have common := ih.1 _ _ _ body (typed_extend domain typed [second] _)
          (canonical_extend domain canonical _ _ _) leftValid (model.body_valid second rightValid)
          built inner innerSound output bodyAccepted
        apply (pair_language model.languages left (.mu second) leftNonempty (by intro h; cases h) output).mpr
        change inputLanguage model.languages left output ∧ model.languages second output
        rw [model.body_exact second rightValid]
        exact common
      · next first _ =>
        obtain ⟨shapes, built, rfl⟩ := map_success _ _ _ completed
        rw [meaning] at accepted
        obtain ⟨⟨shape, member⟩, accepted⟩ := accepted
        obtain ⟨input, present, execution⟩ := BinderSchedule.related_origin _ (walk_exact _ _ _ built) shape member
        have common := ih.1 domain input right shape typed canonical (model.union_valid first leftValid input present)
          rightValid execution environment environmentSound term accepted
        exact ⟨(model.union_exact first leftValid term).mpr ⟨input, present, common.1⟩, common.2⟩
      · next second _ _ =>
        obtain ⟨shapes, built, rfl⟩ := map_success _ _ _ completed
        rw [meaning] at accepted
        obtain ⟨⟨shape, member⟩, accepted⟩ := accepted
        obtain ⟨input, present, execution⟩ := BinderSchedule.related_origin _ (walk_exact _ _ _ built) shape member
        have common := ih.1 domain left input shape typed canonical leftValid (model.union_valid second rightValid input present)
          execution environment environmentSound term accepted
        exact ⟨common.1, (model.union_exact second rightValid term).mpr ⟨input, present, common.2⟩⟩
      · next name _ _ =>
        cases found : lookup domain name with
        | none => rw [found] at completed; cases completed
        | some binder =>
          rw [found] at completed
          have same := typed name binder found
          subst binder
          exact ih.1 domain (.mu name) right result typed canonical (model.reference_valid name leftValid)
            rightValid completed environment environmentSound term accepted
      · next name _ _ _ =>
        cases found : lookup domain name with
        | none => rw [found] at completed; cases completed
        | some binder =>
          rw [found] at completed
          have same := typed name binder found
          subst binder
          exact ih.1 domain left (.mu name) result typed canonical leftValid (model.reference_valid name rightValid)
            completed environment environmentSound term accepted
      · next first second =>
        obtain ⟨shapes, built, rfl⟩ := map_success _ _ _ completed
        rw [meaning] at accepted
        obtain ⟨⟨shape, member⟩, accepted⟩ := accepted
        obtain ⟨rows, present, execution⟩ := BinderSchedule.related_origin _ (walk_exact _ _ _ built) shape member
        obtain ⟨firstRow, firstMember, secondRow, secondMember, labels, rfl⟩ :=
          (SymbolicHashJoin.hashJoin_member Row.label (fun a b => (a, b)) _ _ rows).mp present
        obtain ⟨terms, same, firstChildren, secondChildren, firstGuard, secondGuard⟩ :=
          ih.2 domain firstRow secondRow shape typed canonical (model.row_valid first leftValid firstRow firstMember)
            (model.row_valid second rightValid secondRow secondMember) execution environment environmentSound term accepted
        exact ⟨(model.row_exact first leftValid term).mpr ⟨firstRow, firstMember, terms, same, firstChildren, firstGuard⟩,
          (model.row_exact second rightValid term).mpr ⟨secondRow, secondMember, terms, by rw [← labels]; exact same, secondChildren, secondGuard⟩⟩
      · cases completed; simp [meaning] at accepted
      · cases completed; simp [meaning] at accepted
    · intro domain left right result typed canonical leftValid rightValid completed environment environmentSound term accepted
      rw [edge.eq_def] at completed
      dsimp only at completed
      split at completed
      · cases completed; simp [meaning] at accepted
      · next arity =>
        obtain ⟨shapes, built, rfl⟩ := map_success _ _ _ completed
        rw [meaning] at accepted
        obtain ⟨terms, same, children, guards⟩ := accepted
        have paired := SymbolicInternedView.all₂_scoped_imp _ _ _ _ (walk_exact _ _ _ built) (by
          intro inputs member shape _ execution
          have originals := List.of_mem_zip member
          exact ih.1 domain inputs.1 inputs.2 shape typed canonical (leftValid inputs.1 originals.1)
            (rightValid inputs.2 originals.2) execution environment environmentSound)
        have common := forest_common model.languages environment left.children right.children shapes (by omega) paired terms children
        exact ⟨terms, same, common.1, common.2, guards⟩

/-- Every accepted term of a completed closed-domain output belongs to both original input languages. -/
theorem top_sound [DecidableEq Symbol] (view : View Symbol) (model : InputMeaning view) (fuel : Nat)
    (left right : Input) (leftValid : model.valid left) (rightValid : model.valid right)
    (result : Shape (Payload Symbol)) (completed : run view fuel {} left right = .ok result)
    (environment : Env Symbol) (term : Term Symbol) (accepted : meaning environment result term) :
    inputLanguage model.languages left term ∧ inputLanguage model.languages right term :=
  (run_sound view model fuel).1 {} left right result (by intro name input stored; cases stored)
    (by intro key member; cases member) leftValid rightValid completed environment
    (by intro key member; cases member) term accepted
