theorem current_stored_universe_auxiliary_cutoff_queries
    [DecidableEq Symbol] [DecidableEq (EdgeKey Symbol Guard)] {width : Nat}
    (lib : Library) (query : Query) (signed : Int) (whole : Expression)
    (accepted : SourceSynthesisCurrentConstruction.firstOrderRaw lib query signed = .ok whole)
    (rootValue : Value Symbol Guard) (sourceHeap : InternedConstructorTemplates.State Symbol Guard)
    (built : InternedFreshOrdinaryConstruction.build (ordinarySymbols whole) [] empty =
      .done rootValue sourceHeap)
    (key : NodeKey) (entry : Interned NodeKey) (body depth ordinary : Value Symbol Guard)
    (stored : sourceHeap.nodeRecords key = some (.mu entry body depth ordinary))
    (heap : InternedConstructorTemplates.State Symbol Guard) (facts : Facts heap) (growth : Grows sourceHeap heap)
    (domains : Cache.State IntersectionFlatMemoKeyDomain.DomainKey) (cacheValid : Cache.Valid domains)
    (pattern : Value Symbol Guard) (_patternAvailable : Available heap pattern)
    (_patternClosed : pattern.free = []) (patternId : Nat)
    (_patternIdentity : identity pattern = some patternId)
    (_patternKind : tag pattern = 1 ∨ tag pattern = 3) :
    ∃ (bodyEntry : Interned NodeKey) (bodyRows : List (EdgeValue Symbol Guard)),
      body = .node bodyEntry bodyRows 0 [.interned entry.identity] ∧
      let universeValue : Value Symbol Guard := .mu entry body depth ordinary
      let referenceValue : Value Symbol Guard := .reference (.interned entry.identity)
      let original := SourceRecursiveIntersectionEnvironment.enter domains IntersectionEmptyDomain.emptyKey
        universeValue pattern entry.identity patternId
      let bodyFirst := SourceRecursiveIntersectionEnvironment.enter original.2 original.1.key
        body universeValue bodyEntry.identity entry.identity
      let bodyTerminal := SourceRecursiveIntersectionEnvironment.enter bodyFirst.2 bodyFirst.1.key
        universeValue referenceValue entry.identity entry.identity
      let diagonalFirst := SourceRecursiveIntersectionEnvironment.enter original.2 original.1.key
        universeValue referenceValue entry.identity entry.identity
      let diagonalTerminal := SourceRecursiveIntersectionEnvironment.enter diagonalFirst.2 diagonalFirst.1.key
        body universeValue bodyEntry.identity entry.identity
      ∀ (request : NodeRequest Symbol Guard) (cutoff : Nat × Nat),
        ((request = ⟨bodyTerminal.1, body, referenceValue⟩ ∧ cutoff = pair bodyEntry.identity entry.identity) ∨
          (request = ⟨diagonalTerminal.1, referenceValue, referenceValue⟩ ∧
            cutoff = pair entry.identity entry.identity)) →
        decision request.domain.key request.left request.right = .repeated cutoff ∧
        compileNode request = .pure (repeatedValue cutoff) ∧
        ∀ (Output : Type) (next : Value Symbol Guard → Code Symbol Guard Output)
          (before after : Checked width) (fuel : Nat) (value : Output),
          SourceCurrentAllocatedNativePhysicalCache.Valid before.native →
          NodeReady before.native.machine request →
          (∀ key, SourceRecursiveIntersectionMemo.requestKey width request.domain request.left request.right = some key →
            SourceRecursiveIntersectionMemo.KeySafe width request.domain request.left request.right) →
          Returned (fun old result => old = request → result = repeatedValue cutoff) before.native.nodes →
          execute fuel (.code (.node request next)) before = .done value after →
          ∃ (total inner outer : Nat) (middle : Checked width),
            total < fuel ∧ inner + outer = total ∧
            Steps inner (.code (requestNode request.domain request.left request.right)) before
              (.code (.pure (repeatedValue cutoff))) middle ∧
            Steps outer (.code (next (repeatedValue cutoff))) middle (.code (.pure value)) after

    [DecidableEq Symbol] [DecidableEq (EdgeKey Symbol Guard)]
    (sourceName cutoffLeft cutoffRight : Nat)
    (request : EdgeRequest Symbol Guard) (before after : Checked width)
    (fuel traceFuel : Nat) (result : EdgeValue Symbol Guard)
    (physical : Physical before.native)
    (self : request.right = request.left) (sourceTop : request.left.entry.key.constraint = top)
    (valid : SourceEqualityConstraintMemoAllocation.Valid before.operations)
    (fieldOrigin : SourceEqualityOriginalFieldOrigins.Origin before.operations.constraints
      request.left.entry.key.constraint)
    (returned : Returned (fun old row => old = request →
      row.children = request.left.children.map
        (input (replacement (.interned sourceName)
          (repeatedValue (SourceRecursiveIntersectionControl.pair cutoffLeft cutoffRight)))) ∧
      row.entry.key = ⟨request.left.entry.key.symbol,
        (request.left.children.map (input (replacement (.interned sourceName)
          (repeatedValue (SourceRecursiveIntersectionControl.pair cutoffLeft cutoffRight))))).map Value.referenceOf, top⟩)
      before.native.edges)
    (done : execute fuel (.code (requestEdge request.domain request.left request.right)) before = .done result after)
    (actual : Trace traceFuel (.code (requestEdge request.domain request.left request.right)) before
      (.code (.pure result)) after)
    (childrenReplay : ∀ stepsBefore childFuel, 0 < stepsBefore → stepsBefore + childFuel ≤ fuel → childFuel < fuel →
      ∀ pair ∈ request.left.children.zip request.right.children,
      ∀ (next : Value Symbol Guard → Code Symbol Guard (EdgeValue Symbol Guard))
        (first last : Checked width) (value : Value Symbol Guard),
      Steps stepsBefore (.code (requestEdge request.domain request.left request.right)) before
        (.code ((requestNode request.domain pair.1 pair.2).bind next)) first →
      execute childFuel (.code (requestNode request.domain pair.1 pair.2)) first = .done value last →
      value = input (replacement (.interned sourceName)
        (repeatedValue (SourceRecursiveIntersectionControl.pair cutoffLeft cutoffRight))) pair.1) :
    result.children = request.left.children.map
      (input (replacement (.interned sourceName)
        (repeatedValue (SourceRecursiveIntersectionControl.pair cutoffLeft cutoffRight)))) ∧
    result.entry.key = ⟨request.left.entry.key.symbol,
      (request.left.children.map (input (replacement (.interned sourceName)
        (repeatedValue (SourceRecursiveIntersectionControl.pair cutoffLeft cutoffRight))))).map Value.referenceOf, top⟩

      .done rootValue sourceHeap)

      (.code (.enter IntersectionEmptyDomain.emptyDomain (.mu entry body depth ordinary)
        pattern entry.identity patternId entryNext)) entryBefore)
    (operand : Node) (operandAvailable : Available before.native.machine.heap operand)
    (operandClosed : operand.free = []) (smaller : height operand < height pattern)

      let entered := SourceRecursiveIntersectionEnvironment.enter entryBefore.native.machine.domains
        IntersectionEmptyDomain.emptyKey (.mu entry body depth ordinary) pattern entry.identity patternId

        if referenceFirst then .reference (.interned entry.identity) else operand,
        if referenceFirst then operand else .reference (.interned entry.identity)⟩).bind next)) before) :

    let entered := SourceRecursiveIntersectionEnvironment.enter entryBefore.native.machine.domains
      IntersectionEmptyDomain.emptyKey universeValue pattern entry.identity patternId

    Trace entryCount initial start
      (.code (.enter IntersectionEmptyDomain.emptyDomain universeValue
        pattern entry.identity patternId entryNext)) entryBefore ∧

      (.code (.enter IntersectionEmptyDomain.emptyDomain universeValue
