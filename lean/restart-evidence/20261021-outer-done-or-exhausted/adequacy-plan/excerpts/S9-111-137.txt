  111 /-- A completed actual node trace derives its own result mass and numeric headroom from the original heap. -/
  112 theorem node_return (request : NodeRequest Symbol Guard) {before after : Checked width} (value : Value Symbol Guard)
  113     (actual : Trace fuel (.code (requestNode request.domain request.left request.right)) before (.code (.pure value)) after)
  114     (good : Good before) (ready : NodeReady before.native.machine request) :
  115     execute (fuel + 1) (.code (requestNode request.domain request.left request.right)) before = .done value after ∧
  116       Stored (massAfter fuel (initialBound before.native.machine.heap)) after.native.machine.heap ∧
  117       InternedValueMass.value value ≤ massAfter fuel (initialBound before.native.machine.heap) ∧
  118       after.native.machine.heap.nodes.next ≤ before.native.machine.heap.nodes.next + credit fuel (initialBound before.native.machine.heap) ∧
  119       after.native.machine.heap.edges.next ≤ before.native.machine.heap.edges.next + credit fuel (initialBound before.native.machine.heap) ∧

  121   obtain ⟨_widths, available, _good, _fields, stored, nodes, edges, domains⟩ := node_prefix request actual good ready
  122   exact ⟨trace_done _ _ _ _ _ actual.steps, stored,
  123     InternedValueMassHeap.available after.native.machine.heap stored value available, nodes, edges, domains⟩
