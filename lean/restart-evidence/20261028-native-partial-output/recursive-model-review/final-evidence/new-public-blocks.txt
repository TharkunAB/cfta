def MicroCFTA.SourcePruningEnumerationSelectedValue.run : {Symbol Guard : Type} →
  [DecidableEq (MicroCFTA.InternedConstructorTemplates.EdgeKey Symbol Guard)] →
    Nat →
      MicroCFTA.InternedConstructorTemplates.Value Symbol Guard →
        MicroCFTA.InternedConstructorTemplates.State Symbol Guard →
          MicroCFTA.MergedSourceComparison.Result Symbol Guard
            (MicroCFTA.InternedConstructorTemplates.Value Symbol Guard) :=
fun {Symbol Guard} [DecidableEq (MicroCFTA.InternedConstructorTemplates.EdgeKey Symbol Guard)] fuel value before =>
  match value with
  | value@h:(MicroCFTA.InternedConstructorTemplates.Value.mu entry body depthShape ordinaryShape) =>
    match MicroCFTA.InternedFreshOrdinaryStoredSubstitution.unfold value before with
    | MicroCFTA.InternedNestedConstruction.Outcome.done result after =>
      MicroCFTA.MergedSourceComparison.Result.done result after
    | MicroCFTA.InternedNestedConstruction.Outcome.pending key after =>
      MicroCFTA.MergedSourceComparison.Result.failed (MicroCFTA.MergedSourceComparison.Fault.pending key) after
  | value@h:(MicroCFTA.InternedConstructorTemplates.Value.union entry members depth free) =>
    match MicroCFTA.MergedFreshOrdinaryExposure.nodeEdges fuel value before with
    | MicroCFTA.MergedSourceExposure.Result.done rows middle =>
      have rebuilt :=
        MicroCFTA.InternedConstructorTemplates.run (MicroCFTA.InternedConstructorTemplates.makeNode rows) middle;
      MicroCFTA.MergedSourceComparison.Result.done rebuilt.fst rebuilt.snd
    | MicroCFTA.MergedSourceExposure.Result.unresolved name after =>
      MicroCFTA.MergedSourceComparison.Result.failed (MicroCFTA.MergedSourceComparison.Fault.unresolved name) after
    | MicroCFTA.MergedSourceExposure.Result.pending key after =>
      MicroCFTA.MergedSourceComparison.Result.failed (MicroCFTA.MergedSourceComparison.Fault.pending key) after
    | MicroCFTA.MergedSourceExposure.Result.exhausted after =>
      MicroCFTA.MergedSourceComparison.Result.failed MicroCFTA.MergedSourceComparison.Fault.exhausted after
  | x => MicroCFTA.MergedSourceComparison.Result.done value before
