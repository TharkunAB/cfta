/-- A successful partial render preserves the complete compression frame. -/
theorem expandPartial_frame (graph : Graph Symbol Guard) (fuel : Nat)
    (fragment : Fragment Symbol) (before after : State Symbol Guard)
    (term : Term (PartialSymbol Symbol)) (total : UnionFind.Total before.representatives)
    (success : Success (expandPartial graph fuel fragment) before term after) :
    Frame before after := by
  induction fuel generalizing fragment before after term with
  | zero => exact (success_failure .fuelExhausted before after term success).elim
  | succ fuel ih =>
      cases fragment with
      | node symbol children =>
          change Success (children.mapM (expandPartial graph fuel) >>= fun terms =>
            pure (.node (.concrete symbol) terms)) before term after at success
          obtain ⟨terms, middle, traversed, final⟩ := (success_bind _ _ _ _ _).mp success
          have step : ∀ child ∈ children, ∀ previous output next, Frame before previous →
              Success (expandPartial graph fuel child) previous output next → Frame before next := by
            intro child _ previous output next previousFrame childSuccess
            exact previousFrame.trans
              (ih child previous next output (previousFrame.total total) childSuccess)
          have rendered := mapM_invariant (expandPartial graph fuel) (Frame before) children step
            before middle terms (.refl before) traversed
          exact rendered.trans (pure_frame (.node (.concrete symbol) terms) term middle after final)
      | hole uv =>
          change Success (getValue (fuel + 1) uv >>= fun value => match value with
            | .enumerated stored => expandPartial graph fuel stored
            | .unenumerated (some node) [] => match graph.node node with
              | .recursive _ => pure (.node .truncated [])
              | _ => pure (.node (.hole uv) [])
            | _ => pure (.node (.hole uv) [])) before term after at success
          obtain ⟨value, middle, reading, final⟩ := (success_bind _ _ _ _ _).mp success
          have readFrame := (getValue_frame (fuel + 1) uv before middle value total reading).1
          cases value with
          | eliminated => exact readFrame.trans (pure_frame _ term middle after final)
          | enumerated stored =>
              exact readFrame.trans (ih stored middle after term (readFrame.total total) final)
          | unenumerated node scs =>
              cases node with
              | none => exact readFrame.trans (pure_frame _ term middle after final)
              | some node =>
                  cases scs with
                  | cons first rest => exact readFrame.trans (pure_frame _ term middle after final)
                  | nil =>
                      cases view : graph.node node <;> simp only [view] at final <;>
                        exact readFrame.trans (pure_frame _ term middle after final)

/-- More fuel retains the exact successful partial term and final state. -/
theorem expandPartial_moreFuel (graph : Graph Symbol Guard) (fuel extra : Nat)
    (fragment : Fragment Symbol) :
    Retains (expandPartial graph fuel fragment) (expandPartial graph (fuel + extra) fragment) := by
  induction fuel generalizing fragment with
  | zero =>
      intro before value after success
      exact (success_failure .fuelExhausted before after value success).elim
  | succ fuel ih =>
      rw [Nat.succ_add]
      cases fragment with
      | node symbol children =>
          simp only [expandPartial]
          exact retains_bind _ _ _ _ (retains_mapM _ _ ih children) (fun _ => retains_refl _)
      | hole uv =>
          simp only [expandPartial]
          apply retains_bind _ _ _ _ (by simpa [Nat.add_assoc, Nat.add_comm, Nat.add_left_comm] using getValue_moreFuel (fuel + 1) extra uv)
          intro value
          cases value with
          | eliminated => exact retains_refl _
          | enumerated stored => exact ih stored
          | unenumerated node scs => cases node <;> cases scs <;> exact retains_refl _
