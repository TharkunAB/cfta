    (graph : Graph Symbol Constraint) (templates : List (Template.Pattern Symbol))

    (order : Metadata Typing → List Nat → Option Nat) (metadata updated : Metadata Typing)

      metadata workFuel workFuel workFuel workFuel workFuel workFuel uv caller before memos.query =

      Packet formulas solver inventory caller memos updated
        (EnumerationExpandableIndex.frame core parked) ∧
      Fits profile caller.checked ∧
      SourceEqualityAnnotationHeap.Heap property caller.checked.native.machine.heap

theorem two_actual_parked_prefixes_packet_held_history
    (holeKey : Typing → Nat → Key) (fits : Key → Symbol → Bool) (admits : Key → Node → Bool)
    (graph : Graph Symbol Constraint) (templates : List (Template.Pattern Symbol))
    (checkEmpty : Bool) (edgesOf : Node → List (EdgeValue Symbol Constraint))
    (stop : Option (Metadata Typing → Nat → Bool))
    (order : Metadata Typing → List Nat → Option Nat)
    (formulas : SourceGuards.Formulas FullSyntax.Formula)
    (solver : SourceGuards.Entailment FullSyntax.Formula) (inventory : Inventory)
    (caller : Caller width) (memos : Memos) (before firstParked secondParked : Cached)
    (metadata firstUpdated secondUpdated : Metadata Typing)
    (profile : Profile) (property : Constraint → Prop)
    (packet : Packet formulas solver inventory caller memos metadata before)
    (bounded : Fits profile caller.checked)
    (known : SourceEqualityAnnotationHeap.Heap property caller.checked.native.machine.heap)
    (workFuel depth firstUv secondUv : Nat)
    (firstChecked firstCore secondChecked secondCore : Core) (next : Option Next)
    (chosen : SourcePruningEnumerationRestart.decision stop order metadata before = some (.expand firstUv))
    (firstActual : SourcePruningEnumerationSelectedStep.runPrefix holeKey fits admits graph templates stop order
      metadata workFuel workFuel workFuel workFuel workFuel workFuel firstUv caller before memos.query =
        .parked firstChecked (some firstParked) (some [.ok ((firstUpdated, some (.expand secondUv)), firstCore)]))
    (secondActual : SourcePruningEnumerationSelectedStep.runPrefix holeKey fits admits graph templates stop order
      firstUpdated workFuel workFuel workFuel workFuel workFuel workFuel secondUv caller
      (EnumerationExpandableIndex.frame firstCore firstParked) memos.query =
        .parked secondChecked (some secondParked) (some [.ok ((secondUpdated, next), secondCore)]))
    (index : Nat) (output : Output width Typing)
    (observed : (emitted (resumeWithNarrow
      (SourcePruningEnumerationSharedEagerChildren.eagerChild
        checkEmpty workFuel workFuel workFuel workFuel workFuel edgesOf)
      (SourcePruningOuterNarrow.narrow holeKey fits admits)
      graph templates checkEmpty edgesOf stop order workFuel (depth + 2)
      metadata caller memos.query before (some (.expand firstUv))))[index]? = some output) :
    ∃ (handle : Nat) (bucket : List EnumerationState.Suspended) (value : Node),
      before.core.values[firstUv]? = some (.unenumerated (some handle) bucket) ∧
      SourcePruningEnumerationNodes.lookupNode caller.checked.native.machine.heap handle = some value ∧
      MergedOrdinaryHeight.recursiveBoundary value = true ∧
      Packet formulas solver inventory caller memos firstUpdated
        (EnumerationExpandableIndex.frame firstCore firstParked) ∧
      Packet formulas solver inventory caller memos secondUpdated
        (EnumerationExpandableIndex.frame secondCore secondParked) ∧
      some (.expand secondUv) = SourcePruningEnumerationRestart.decision stop order firstUpdated
        (EnumerationExpandableIndex.frame firstCore firstParked) ∧
      next = SourcePruningEnumerationRestart.decision stop order secondUpdated
        (EnumerationExpandableIndex.frame secondCore secondParked) ∧
      secondUv ≠ firstUv ∧
      HeldFragmentHistory firstUv handle firstParked.core
        (EnumerationExpandableIndex.frame secondCore secondParked).core ∧
      (emitted (resumeWithNarrow
        (SourcePruningEnumerationSharedEagerChildren.eagerChild
          checkEmpty workFuel workFuel workFuel workFuel workFuel edgesOf)
        (SourcePruningOuterNarrow.narrow holeKey fits admits)
        graph templates checkEmpty edgesOf stop order workFuel depth
        secondUpdated caller memos.query (EnumerationExpandableIndex.frame secondCore secondParked) next))[index]? =
          some output
