/-- Complete absent original-row pairs and retain the same stable checked boundary. -/
theorem both_miss_complete [DecidableEq Symbol] [DecidableEq (EdgeKey Symbol Guard)]
    (bits : Nat) (source : InternedConstructorTemplates.State Symbol Guard)
    (facts : InternedNestedFacts.Facts source)
    (formulas : SourceGuards.Formulas FullSyntax.Formula)
    (solver : SourceGuards.Entailment FullSyntax.Formula)
    (inputModel : InputMeaning (view source (guardHolds formulas solver)))
    (inputRanked : RankedInput (view source (guardHolds formulas solver)) inputModel.valid)
    (leftParent rightParent : Value Symbol Guard)
    (leftOriginal : OriginalInput source (inputOf leftParent))
    (rightOriginal : OriginalInput source (inputOf rightParent))
    (leftAvailable : Available source leftParent) (rightAvailable : Available source rightParent)
    (leftClosed : leftParent.free = []) (rightClosed : rightParent.free = [])
    (left right : Edge)
    (leftMember : left ∈ nodeEdges leftParent) (rightMember : right ∈ nodeEdges rightParent)
    (symbols : left.entry.key.symbol = right.entry.key.symbol)
    (before : Wrapper (bits + 2))
    (stable : Stable (view source (guardHolds formulas solver)) inputModel inputRanked
      (guardHolds formulas solver) before)
    (missing : before.memo.table (left.entry.identity, right.entry.identity) = none)
    (reverseMissing : before.memo.table (right.entry.identity, left.entry.identity) = none)
    (good : Good before.checked)
    (contextual : SourceCurrentAllocatedNativeContextCache.Contextual
      (view source (guardHolds formulas solver)) inputModel inputRanked
      (guardHolds formulas solver) before.checked.native)
    (semantic : SemanticReady before.checked.native.machine.heap)
    (growth : Grows source before.checked.native.machine.heap)
    (history : Extends source.nodes.issued before.checked.native.machine.heap.nodes.issued)
    (idle : NativeIdle before.checked.native)
    (room : SourceCurrentAllocatedNativeResourceForecast.Room bits
      ⟨source.nodes.next, source.edges.next, before.checked.native.machine.domains.next, 0⟩) :
    let ordered := if right.entry.identity < left.entry.identity then (right, left) else (left, right)
    let request : EdgeRequest Symbol Guard := ⟨IntersectionEmptyDomain.emptyDomain, ordered.1, ordered.2⟩
    ∃ (fuel nativeFuel handle : Nat) (result : Edge) (after : Wrapper (bits + 2)),
      SourceCurrentSameSymbolEdgeControl.execute fuel (.call (left, right) []) before = .done result after ∧
      SourceCurrentAllocatedNativeContextualCompletion.Receipt
        (semanticView := view source (guardHolds formulas solver)) (model := inputModel)
        (ranked := inputRanked) (semanticHolds := guardHolds formulas solver)
        nativeFuel (.edgeBody request .pure) before.checked result after.checked
        (fun final value => EdgeResident final.heap value)
        (fun _ value => SourceCurrentAllocatedNativeContextCache.EdgeLaw
          (view source (guardHolds formulas solver)) inputModel inputRanked
          (guardHolds formulas solver) request value) ∧
      Stable (view source (guardHolds formulas solver)) inputModel inputRanked
        (guardHolds formulas solver) after ∧
      NativeIdle after.checked.native ∧
      Extends before.memo.table after.memo.table ∧
      FlatMemoThunk.Returned before.memo after.memo ∧
      after.memo.table (left.entry.identity, right.entry.identity) = some handle ∧
      after.memo.cells handle = some ⟨(left, right), .returned result⟩ ∧
      (right.entry.identity < left.entry.identity →
        ∃ reversedHandle, reversedHandle ≠ handle ∧
          after.memo.table (right.entry.identity, left.entry.identity) = some reversedHandle ∧
          after.memo.cells reversedHandle = some ⟨(right, left), .returned result⟩) := by
  dsimp only
  let ordered := if right.entry.identity < left.entry.identity then (right, left) else (left, right)
  let request : EdgeRequest Symbol Guard := ⟨IntersectionEmptyDomain.emptyDomain, ordered.1, ordered.2⟩
  have sourceScope := rows_scope source facts leftParent rightParent leftOriginal rightOriginal
    leftAvailable rightAvailable left right leftMember rightMember
  have orderedInputs : EdgeScope source request ∧
      Completes (view source (guardHolds formulas solver))
        (SourceRecursiveIntersectionNative.edgeSemantic (guardHolds formulas solver) request) ∧
      request.left.entry.key.symbol = request.right.entry.key.symbol := by
    by_cases swap : right.entry.identity < left.entry.identity
    · refine ⟨?_, ?_, ?_⟩
      · simpa only [request, ordered, swap, ↓reduceIte] using
          (show EdgeScope source ⟨IntersectionEmptyDomain.emptyDomain, right, left⟩ from
            ⟨sourceScope.rightResident, sourceScope.leftResident,
              sourceScope.right, sourceScope.left, sourceScope.bindings⟩)
      · simpa only [request, ordered, swap, ↓reduceIte] using
          SourceRecursiveIntersectionOriginalRowCompletion.edge_complete source facts
            (guardHolds formulas solver) rightParent leftParent rightOriginal leftOriginal
            rightAvailable leftAvailable rightClosed leftClosed right left rightMember leftMember
      · simpa only [request, ordered, swap, ↓reduceIte] using symbols.symm
    · refine ⟨?_, ?_, ?_⟩
      · simpa only [request, ordered, swap, ↓reduceIte] using sourceScope
      · simpa only [request, ordered, swap, ↓reduceIte] using
          SourceRecursiveIntersectionOriginalRowCompletion.edge_complete source facts
            (guardHolds formulas solver) leftParent rightParent leftOriginal rightOriginal
            leftAvailable rightAvailable leftClosed rightClosed left right leftMember rightMember
      · simpa only [request, ordered, swap, ↓reduceIte] using symbols
  obtain ⟨scope, complete, orderedSymbols⟩ := orderedInputs
  have ready : EdgeReady before.checked.native.machine request :=
    ⟨empty_domain _, edge_grows source _ growth _ scope.leftResident,
      edge_grows source _ growth _ scope.rightResident⟩
  have ranges := SourceCurrentAllocatedDirectEdgeWordRanges.edge_body bits source before.checked
    facts good semantic growth history request.left request.right scope ready room
  have active : SourceRecursiveIntersectionNative.ActiveAbove
      (view source (guardHolds formulas solver)) (guardHolds formulas solver)
      (requiredFuel (view source (guardHolds formulas solver))
        (SourceRecursiveIntersectionNative.edgeSemantic (guardHolds formulas solver) request) complete)
      before.checked.native := idle.active _ _ _ _
  have memoValid : FlatMemoThunk.Valid
      (fun (pair : Edge × Edge) (key : Nat × Nat) =>
        (pair.1.entry.identity, pair.2.entry.identity) = key)
      (fun _ => True) before.memo :=
    FlatMemoThunk.Valid.map before.memo _ _ _ stable.valid (fun _ _ => True.intro)
  obtain ⟨fuel, nativeFuel, handle, result, after, done, receipt, stored, cell, _, reversed⟩ :=
    SourceCurrentSameSymbolEdgeForwardCompletion.both_miss_complete source facts formulas solver
      inputModel inputRanked left right before memoValid missing reverseMissing scope complete
      good contextual semantic orderedSymbols ready growth history active ranges
  obtain ⟨afterStable, tables, retained⟩ :=
    SourceCurrentSameSymbolEdgeStable.completed_preserves
      (view source (guardHolds formulas solver)) inputModel inputRanked (guardHolds formulas solver)
      fuel left right result before after stable
      (edge_grows source _ growth _ sourceScope.leftResident)
      (edge_grows source _ growth _ sourceScope.rightResident)
      done receipt.extension receipt.result receipt.law
  have afterIdle : NativeIdle after.checked.native :=
    SourceCurrentAllocatedNativeRunning.done_idle (nativeFuel + 1) (.edgeBody request .pure)
      before.checked after.checked result (by intro value; trivial)
      good.native.shape idle receipt.done
  exact ⟨fuel, nativeFuel, handle, result, after, done, receipt, afterStable, afterIdle,
    tables, retained, stored, cell, reversed⟩
