def MicroCFTA.EnumerationFlatNativeIncomingCache.adapt : MicroCFTA.InternedConstructorTemplates.State
    MicroCFTA.SourcePruningTemplates.Symbol (MicroCFTA.SourceSymbolicIndex.Constraint MicroCFTA.FullSyntax.Formula) →
  Nat →
    List MicroCFTA.EnumerationState.Suspended →
      MicroCFTA.EnumerationFlatCached.Cached MicroCFTA.SourcePruningTemplates.Symbol
          (MicroCFTA.SourceSymbolicIndex.Constraint MicroCFTA.FullSyntax.Formula) →
        MicroCFTA.EnumerationFlatState.State MicroCFTA.SourcePruningTemplates.Symbol
            (MicroCFTA.SourceSymbolicIndex.Constraint MicroCFTA.FullSyntax.Formula) →
          List
              (Except MicroCFTA.EnumerationState.Error
                (Unit ×
                  MicroCFTA.EnumerationFlatState.State MicroCFTA.SourcePruningTemplates.Symbol
                    (MicroCFTA.SourceSymbolicIndex.Constraint MicroCFTA.FullSyntax.Formula))) →
            Option
              (MicroCFTA.EnumerationFlatCached.Cached MicroCFTA.SourcePruningTemplates.Symbol
                (MicroCFTA.SourceSymbolicIndex.Constraint MicroCFTA.FullSyntax.Formula)) :=
fun heap target incomingScs before read raw =>
  match raw with
  | [] =>
    match read.values[target]? with
    | some (MicroCFTA.EnumerationState.UVarValue.unenumerated _oldNode oldScs) =>
      have value := MicroCFTA.EnumerationState.UVarValue.unenumerated (some 0) (incomingScs ++ oldScs);
      have core :=
        { counter := read.counter, representatives := read.representatives, values := read.values.set target value,
          obligations := read.obligations,
          recursionAncestry := fun other =>
            if other = target then (read.expansionAncestry ++ read.recursionAncestry target).eraseDups
            else read.recursionAncestry other,
          expansionAncestry := read.expansionAncestry };
      Option.map (MicroCFTA.EnumerationFlatCached.frame core)
        (MicroCFTA.EnumerationFlatNativeIndex.write heap target value incomingScs []
          (MicroCFTA.EnumerationFlatCached.frame read before))
    | x => none
  | [Except.ok (PUnit.unit, returned)] =>
    returned.values[target]?.bind fun slotValue =>
      Option.map (MicroCFTA.EnumerationFlatCached.frame returned)
        (MicroCFTA.EnumerationFlatNativeIndex.write heap target slotValue incomingScs []
          (MicroCFTA.EnumerationFlatCached.frame read before))
  | x => none
