
/-- One actual checked intersection retains its complete returned state. -/
inductive Step [DecidableEq Symbol] [DecidableEq (EdgeKey Symbol Guard)]
    (left right : Value Symbol Guard) (before : Checked width) :
    Value Symbol Guard → Checked width → Prop where
  | run {fuel : Nat}
      (trace : Trace fuel (.code (requestNode IntersectionEmptyDomain.emptyDomain left right)) before
        (.code (.pure value)) after) : Step left right before value after

/-- A raw completed step has a literal finite checked execution. -/
theorem Step.done [DecidableEq Symbol] [DecidableEq (EdgeKey Symbol Guard)]
    {left right value : Value Symbol Guard} {before after : Checked width}
    (actual : Step left right before value after) :
    ∃ fuel, execute fuel (.code (requestNode IntersectionEmptyDomain.emptyDomain left right)) before = .done value after := by
  cases actual with
  | run trace => exact ⟨_, trace_done _ _ before after value trace.steps⟩

/-- Completed steps from one entry return the same value and every state field. -/
theorem Step.unique [DecidableEq Symbol] [DecidableEq (EdgeKey Symbol Guard)]
    {left right first second : Value Symbol Guard} {before afterFirst afterSecond : Checked width}
    (leftStep : Step left right before first afterFirst) (rightStep : Step left right before second afterSecond) :
    first = second ∧ afterFirst = afterSecond := by
  obtain ⟨firstFuel, firstDone⟩ := leftStep.done
  obtain ⟨secondFuel, secondDone⟩ := rightStep.done
  exact SourceCurrentAllocatedNativeColdClosedExact.completed_unique firstFuel secondFuel _ before
    afterFirst afterSecond first second firstDone secondDone

/-- One literal return retains its computed inventory and the same global table bundle. -/
structure Receipt [DecidableEq Symbol] [DecidableEq (EdgeKey Symbol Guard)]
    (formulas : SourceGuards.Formulas FullSyntax.Formula) (solver : SourceGuards.Entailment FullSyntax.Formula)
    (inventory : Inventory) (left right : Value Symbol Guard) (before : Checked width)
    (value : Value Symbol Guard) (after : Checked width) (memos : Memos) : Prop where
  actual : Step left right before value after
  ready : Ready after
  admitted : At formulas solver inventory after
  effects : SourceCurrentConjunctionRowsAllocation.Effects (physical before) (physical after)
  extension : Extension before.native.machine after.native.machine
  available : Available after.native.machine.heap value
  closed : value.free = []
  meaning : ∀ ambient term,
    InternedGuardedLanguage.meaning (SourceEqualityPCSGuard.holds formulas solver) ambient value term ↔
      InternedGuardedLanguage.meaning (SourceEqualityPCSGuard.holds formulas solver) ambient left term ∧
      InternedGuardedLanguage.meaning (SourceEqualityPCSGuard.holds formulas solver) ambient right term
  globals : Valid after.native.machine.heap memos
  events : ∃ events, SourceGlobalUnfold.Runs ⟨before.native.machine.heap, memos.unfolding⟩
    ⟨after.native.machine.heap, memos.unfolding⟩ events ∧ SourceGlobalUnfold.NoGlobal events
