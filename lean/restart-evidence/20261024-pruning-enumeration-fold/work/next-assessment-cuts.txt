-- | Merge the source UVar into the target UVar, intersecting their constraints.
assimilateUvarVal ::
    (Hashable symbol, Typeable symbol) => UVar -> UVar -> EnumerateM symbol ()
assimilateUvarVal uvTarg uvSrc
    | uvTarg == uvSrc = return ()
    | otherwise = do
        values <- gets _uvarValues
        let srcVal = values IntMap.! uvarToInt uvSrc
        let targVal = values IntMap.! uvarToInt uvTarg
        case srcVal of
            UVarEliminated -> return () -- Happens from duplicate constraints
            _ -> do
                checkEmpty <- gets _constrainedRecursion
                let v = intersectUVarValue checkEmpty srcVal targVal
                guard $ not $ hasEmptyContents v
                -- The suspended constraints of the source move to the
                -- target, in the same class, so no count changes.
                writeUVarValue Sequence.Empty Sequence.Empty uvTarg v
                writeUVarValue Sequence.Empty Sequence.Empty uvSrc UVarEliminated
                modify' $ \s -> case IntMap.lookup (uvarToInt uvSrc) (_recursionAncestry s) of
                    Nothing -> s
                    Just ancestry -> s{_recursionAncestry = IntMap.insertWith HashSet.union (uvarToInt uvTarg) ancestry (_recursionAncestry s)}

-- | Intersect a node and inherited constraints into the value for a UVar.
mergeNodeIntoUVarVal ::
    (Hashable symbol, Typeable symbol) =>
    UVar -> Node symbol -> Seq SuspendedConstraint -> EnumerateM symbol ()
mergeNodeIntoUVarVal uv n scs = do
    uv' <- getUVarRepresentative uv
    let idx = uvarToInt uv'
    values <- gets _uvarValues
    checkEmpty <- gets _constrainedRecursion
    let new = intersectUVarValue checkEmpty (UVarUnenumerated (Just n) scs) (values IntMap.! idx)
    -- The slot keeps its suspended constraints and gains these.
    writeUVarValue scs Sequence.Empty uv' new
    inheritAncestry idx
    guard $ not $ hasEmptyContents new

theorem mapM_terminates (action : Nat → α → EnumerateM Symbol Guard β)
    (invariant : State Symbol Guard → Prop)
    (more : ∀ fuel extra input, Retains (action fuel input) (action (fuel + extra) input))
    (step : ∀ input before, invariant before →
      ∃ fuel value after, Success (action fuel input) before value after ∧ invariant after)
    (inputs : List α) (before : State Symbol Guard) (valid : invariant before) :
    ∃ fuel values after, Success (inputs.mapM (action fuel)) before values after ∧ invariant after := by
  induction inputs generalizing before with
  | nil => exact ⟨0, [], before, (success_pure _ _ _ _).mpr ⟨rfl, rfl⟩, valid⟩
  | cons input rest ih =>
      obtain ⟨firstFuel, value, middle, first, middleValid⟩ := step input before valid
      obtain ⟨restFuel, values, after, continued, afterValid⟩ := ih middle middleValid

      rw [List.mapM_cons]

      refine ⟨value, middle, more firstFuel restFuel input _ _ _ first, ?_⟩

      have extended := retains_mapM _ _ (more restFuel firstFuel) rest _ _ _ continued
      simpa [Nat.add_comm] using extended

  if target = source then waiting else

def union (leftFuel rightFuel : Nat) (forest : Forest) (left right : Nat) : Option Forest := do
  let first ← findRoot leftFuel forest left
  let second ← findRoot rightFuel first.forest right
  if first.representative = second.representative then some second.forest
  else if first.size < second.size then
    some (link second.forest first.representative second.representative first.size second.size)
  else some (link second.forest second.representative first.representative second.size first.size)
