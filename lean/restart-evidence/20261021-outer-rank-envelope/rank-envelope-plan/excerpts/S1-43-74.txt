   43 termination_by (fuel, 3)

   45 /-- An ordinary body reserves its row pass and one raw node constructor. -/
   46 noncomputable def body (fuel budget : Nat) (seed : List Guard × Profile) : List Guard × Profile :=
   47   match fuel with
   48   | 0 => seed
   49   | next + 1 => rebuild (rows next budget seed)
   50 termination_by (fuel, 2)
   51 
   52 /-- Rows reserve the head Parent, its descendants and then the original suffix. -/
   53 noncomputable def rows (fuel budget : Nat) (seed : List Guard × Profile) : List Guard × Profile :=
   54   match fuel with
   55   | 0 => seed
   56   | next + 1 => rows next budget (descendants next budget (parent budget seed))
   57 termination_by (fuel, 2)
   58 
   59 /-- Descendants reserve the child pass and one raw edge constructor. -/
   60 noncomputable def descendants (fuel budget : Nat) (seed : List Guard × Profile) : List Guard × Profile :=
   61   rebuild (children fuel budget seed)
   62 termination_by (fuel, 2)
   63 
   64 /-- The child schedule includes the live walk; contradiction can retain entry bounds. -/
   65 noncomputable def children (fuel budget : Nat) (seed : List Guard × Profile) : List Guard × Profile :=
   66   walk fuel budget seed
   67 termination_by (fuel, 1)
   68 
   69 /-- A walk reserves the selected node and then the original suffix at its returned bounds. -/
   70 noncomputable def walk (fuel budget : Nat) (seed : List Guard × Profile) : List Guard × Profile :=
   71   match fuel with
   72   | 0 => seed
   73   | next + 1 => walk next budget (node next budget seed)
   74 termination_by (fuel, 0)
