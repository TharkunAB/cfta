    7 /-!
    8 # A finite numeric envelope over strict count and mass ranks
    9 
   10 Each recursive candidate has a smaller rank. Its seed is the current
   11 numeric accumulator. This module proves only entry retention and a
   12 positive work bound. It does not prove stage coverage or completion.
   13 -/
   14 namespace MicroCFTA.SourceCurrentOuterRankEnvelope
   15 open SourceCurrentAllocatedNativeResourceForecast (Profile Le)
   16 open SourceCurrentInitialSelectionForecast (cover cover_ge maximum)
   17 local notation "Guard" => SourceSymbolicIndex.Constraint FullSyntax.Formula
   18 local notation "Seed" => List Guard × Profile
   19 local notation "Out" => Nat × Nat × Seed
   20 
   21 private def candidates (rank : Nat × Nat) (seed : Seed) : List (Nat × Nat) :=
   22   ((List.range (rank.1 + 1)).flatMap fun a =>
   23     (List.range (seed.2.mass + 1)).map fun b => (a,b)).filter
   24       (fun child => decide (child.1 < rank.1 ∨ (child.1 = rank.1 ∧ child.2 < rank.2)))
   25 
   26 private theorem candidates_lt (rank : Nat × Nat) (seed : Seed) (child : Nat × Nat)
   27     (member : child ∈ candidates rank seed) : Prod.Lex Nat.lt Nat.lt child rank := by
   28   simp only [candidates, List.mem_filter, decide_eq_true_eq] at member
   29   exact Prod.lex_def.mpr member.2
   30 
   31 private def images (rank : Nat × Nat)
   32     (recurse : ∀ child, Prod.Lex Nat.lt Nat.lt child rank → Seed → Out)
   33     (seed : Seed) : List Out :=
   34   (candidates rank seed).attach.map fun child =>
   35     recurse child.val (candidates_lt rank seed child.val child.property) seed
   36 
   37 private def joinSeed (seed : Seed) (outputs : List Out) : Seed :=
   38   (seed.1 ++ outputs.flatMap (fun output => output.2.2.1),
   39     cover (outputs.map fun output => output.2.2.2) seed.2)
   40 
   41 private def slot (all : Seed → List Out) (acc : Out) (_index : Nat) : Out :=
   42   let outputs := all acc.2.2
   43   (acc.1 + 1 + maximum (outputs.map fun output => output.1),
   44     max acc.2.1 (maximum (outputs.map fun output => output.2.1)),
   45     joinSeed acc.2.2 outputs)
   46 
   47 private noncomputable def row (budget : Nat) (all : Seed → List Out)
   48     (acc : Out) (_index : Nat) : Out :=
   49   let post := SourceCurrentOuterInitialSchedule.parent budget acc.2.2
   50   let walked := (List.range post.2.mass).foldl (slot all) (0,0,post)
   51   (acc.1 + 1 + walked.1,
   52     max acc.2.1 (max (SourceCurrentParentUniformEnvelope.fuel budget acc.2.2.1 acc.2.2.2) walked.2.1),
   53     SourceCurrentOuterInitialSchedule.rebuild walked.2.2)
   54 
   55 private noncomputable def recipe (budget : Nat) (rank : Nat × Nat)
   56     (recurse : ∀ child, Prod.Lex Nat.lt Nat.lt child rank → Seed → Out)
   57     (seed : Seed) : Out :=
   58   let completed := (List.range rank.2).foldl (row budget (images rank recurse)) (0,0,seed)
   59   (1 + completed.1, completed.2.1, SourceCurrentOuterInitialSchedule.rebuild completed.2.2)
   60 
   61 /-- Compute finite work, lower fuel, and an output seed from the initial rank and seed. -/
   62 noncomputable def envelope (budget count mass : Nat)
   63     (seed : List Guard × Profile) : Nat × Nat × (List Guard × Profile) :=
   64   WellFounded.fix (C := fun _ : Nat × Nat => Seed → Out)
   65     (Prod.lex Nat.lt_wfRel Nat.lt_wfRel).wf (recipe budget) (count,mass) seed
   66 
   67 private theorem envelope_eq (budget count mass : Nat) (seed : Seed) :
   68     envelope budget count mass seed = recipe budget (count,mass)
   69       (fun child _ current => envelope budget child.1 child.2 current) seed := by
   70   unfold envelope
   71   rw [WellFounded.fix_eq]
   72   apply congrArg (fun recurse => recipe budget (count,mass) recurse seed)
   73   funext child smaller current
   74   cases child
   75   rfl
   76 
   77 private theorem fold_retains (step : Out → Nat → Out)
   78     (growth : ∀ acc index, Le acc.2.2.2 (step acc index).2.2.2 ∧
   79       acc.2.2.1 ⊆ (step acc index).2.2.1)
   80     (values : List Nat) (initial : Out) :
   81     Le initial.2.2.2 (values.foldl step initial).2.2.2 ∧
   82       initial.2.2.1 ⊆ (values.foldl step initial).2.2.1 := by
   83   induction values generalizing initial with
   84   | nil => exact ⟨Le.refl _, fun _ member => member⟩
   85   | cons first rest ih =>
   86       have one := growth initial first
   87       have last := ih (step initial first)
   88       exact ⟨one.1.trans last.1, fun _ member => last.2 (one.2 member)⟩
   89 
   90 private theorem fold_work (step : Out → Nat → Out)
   91     (growth : ∀ acc index, acc.1 + 1 ≤ (step acc index).1)
   92     (values : List Nat) (initial : Out) :
   93     initial.1 + values.length ≤ (values.foldl step initial).1 := by
   94   induction values generalizing initial with
   95   | nil => simp only [List.length_nil, List.foldl_nil, Nat.add_zero, Nat.le_refl]
   96   | cons first rest ih =>
   97       have one := growth initial first
   98       have last := ih (step initial first)
   99       simp only [List.length_cons, List.foldl_cons]
  100       omega

  102 private theorem slot_retains (all : Seed → List Out) (acc : Out) (index : Nat) :
  103     Le acc.2.2.2 (slot all acc index).2.2.2 ∧ acc.2.2.1 ⊆ (slot all acc index).2.2.1 := by
  104   dsimp only [slot, joinSeed]
  105   exact ⟨cover_ge _ _, fun _ member => List.mem_append_left _ member⟩
  106 
  107 private theorem row_retains (budget : Nat) (all : Seed → List Out) (acc : Out) (index : Nat) :
  108     Le acc.2.2.2 (row budget all acc index).2.2.2 ∧
  109       acc.2.2.1 ⊆ (row budget all acc index).2.2.1 := by
  110   dsimp only [row]
  111   generalize post_eq : SourceCurrentOuterInitialSchedule.parent budget acc.2.2 = post
  112   let walked := (List.range post.2.mass).foldl (slot all) (0,0,post)
  113   have one := SourceCurrentOuterInitialSchedule.parent_ge budget acc.2.2
  114   rw [post_eq] at one
  115   have middle : Le post.2 walked.2.2.2 ∧ post.1 ⊆ walked.2.2.1 :=
  116     fold_retains (slot all) (slot_retains all) (List.range post.2.mass) (0,0,post)
  117   have last := SourceCurrentOuterInitialSchedule.rebuild_ge walked.2.2
  118   change Le acc.2.2.2 (SourceCurrentOuterInitialSchedule.rebuild walked.2.2).2 ∧
  119     acc.2.2.1 ⊆ (SourceCurrentOuterInitialSchedule.rebuild walked.2.2).1
  120   refine ⟨one.1.trans (middle.1.trans last.1), ?_⟩
  121   change acc.2.2.1 ⊆ walked.2.2.1
  122   intro guard member
  123   have present : guard ∈ post.1 := one.2 member
  124   exact middle.2 present
  125 
  126 private theorem row_work (budget : Nat) (all : Seed → List Out) (acc : Out) (index : Nat) :
  127     acc.1 + 1 ≤ (row budget all acc index).1 := by
  128   dsimp only [row]
  129   exact Nat.le_add_right _ _
  130 
  131 private theorem recipe_retains (budget : Nat) (rank : Nat × Nat)
  132     (recurse : ∀ child, Prod.Lex Nat.lt Nat.lt child rank → Seed → Out)
  133     (seed : Seed) :
  134     rank.2 + 1 ≤ (recipe budget rank recurse seed).1 ∧
  135       Le seed.2 (recipe budget rank recurse seed).2.2.2 ∧
  136       seed.1 ⊆ (recipe budget rank recurse seed).2.2.1 := by
  137   let all := images rank recurse
  138   let completed := (List.range rank.2).foldl (row budget all) (0,0,seed)
  139   have kept := fold_retains (row budget all) (row_retains budget all) (List.range rank.2) (0,0,seed)
  140   have work := fold_work (row budget all) (row_work budget all) (List.range rank.2) (0,0,seed)
  141   have last := SourceCurrentOuterInitialSchedule.rebuild_ge completed.2.2
  142   have countBound : rank.2 ≤ completed.1 := by
  143     simpa only [List.length_range, Nat.zero_add] using work
  144   change rank.2 + 1 ≤ 1 + completed.1 ∧
  145     Le seed.2 (SourceCurrentOuterInitialSchedule.rebuild completed.2.2).2 ∧ seed.1 ⊆ (SourceCurrentOuterInitialSchedule.rebuild completed.2.2).1
  146   exact ⟨by omega, kept.1.trans last.1, fun _ member => last.2 (kept.2 member)⟩
  147 
  148 /-- The envelope has positive work and retains its complete entry profile and alphabet. -/
  149 theorem retains (budget count mass : Nat) (seed : List Guard × Profile) :
  150     mass + 1 ≤ (envelope budget count mass seed).1 ∧
  151     Le seed.2 (envelope budget count mass seed).2.2.2 ∧
  152     seed.1 ⊆ (envelope budget count mass seed).2.2.1 := by
  153   rw [envelope_eq]
  154   exact recipe_retains budget (count,mass)
  155     (fun child _ current => envelope budget child.1 child.2 current) seed
  156 
  157 end MicroCFTA.SourceCurrentOuterRankEnvelope
