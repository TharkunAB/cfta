def put(data, label, classification='historical input'):
    digest = sha(data)
    suffix = '.patch' if label.endswith('.patch') else '.txt'
    path = objects / (digest + suffix)
    if path.exists():
        assert path.read_bytes() == data
    else:
        path.write_bytes(data)
    ref = {'label': label, 'object': str(path.relative_to(packet)), 'sha256': digest,
           'bytes': len(data), 'classification': classification}
    manifest['files'].append(ref)
    return ref

def file(path, label=None, classification='historical input'):
    return put(path.read_bytes(), label or str(path), classification)

def delta(before, after, label):
    result = subprocess.run(['diff', '-U0', '--label', 'a/source', '--label', 'b/source',
                             str(before), str(after)], capture_output=True)
    assert result.returncode == 1
    ref = put(result.stdout, label + '.patch', 'exact GNU zero-context delta')
    item = {'before_sha256': sha(before.read_bytes()), 'after_sha256': sha(after.read_bytes()),
            'patch': ref['object'], 'label': label}
    manifest['deltas'].append(item)
    return item
