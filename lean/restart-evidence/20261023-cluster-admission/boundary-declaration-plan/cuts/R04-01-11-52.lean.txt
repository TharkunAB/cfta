/-- Derive admission for every captured pair after the same completed comparison. -/
theorem completed_admitted
    [DecidableEq Symbol] [DecidableEq (EdgeKey Symbol Guard)]
    {width : Nat} (view : View Symbol) (model : InputMeaning view)
    (ranked : RankedInput view model.valid) (holds : Guard → Term Symbol → Bool)
    (left right result : Edge) (before after : Wrapper width) (fuel : Nat)
    (stable : SourceCurrentSameSymbolEdgeStable.Stable view model ranked holds before)
    (admitted : ∀ (handle : Nat) (pair : Edge × Edge) (value : Edge),
      before.memo.cells handle = some ⟨pair, .returned value⟩ →
      let ordered := if pair.2.entry.identity < pair.1.entry.identity then (pair.2, pair.1) else pair
      SourceCurrentAllocatedNativeContextCache.EdgeInputs view model
        ⟨IntersectionEmptyDomain.emptyDomain, ordered.1, ordered.2⟩)
    (known : SourceCurrentAllocatedNativeContextCache.EdgeInputs view model
      ⟨IntersectionEmptyDomain.emptyDomain, left, right⟩)
    (done : SourceCurrentSameSymbolEdgeControl.execute fuel (.call (left, right) []) before =
      .done result after) :
    SourceCurrentAllocatedNativeAdmissionCache.Requests
      (fun pair : Edge × Edge => SourceCurrentAllocatedNativeContextCache.EdgeInputs view model
        ⟨IntersectionEmptyDomain.emptyDomain, pair.1, pair.2⟩) after.memo ∧
    ∀ (handle : Nat) (pair : Edge × Edge) (value : Edge),
      after.memo.cells handle = some ⟨pair, .returned value⟩ →
      let ordered := if pair.2.entry.identity < pair.1.entry.identity then (pair.2, pair.1) else pair
      SourceCurrentAllocatedNativeContextCache.EdgeInputs view model
        ⟨IntersectionEmptyDomain.emptyDomain, ordered.1, ordered.2⟩ := by
  let property : Edge × Edge → Prop := fun pair =>
    SourceCurrentAllocatedNativeContextCache.EdgeInputs view model
      ⟨IntersectionEmptyDomain.emptyDomain, pair.1, pair.2⟩

  have swapped : ∀ pair : Edge × Edge, property pair → property (pair.2, pair.1) := by
    intro pair inputs
    exact ⟨inputs.2, inputs.1⟩

  have initial : SourceCurrentAllocatedNativeAdmissionCache.Requests property before.memo := by
    intro handle cell stored
    rcases cell with ⟨pair, status⟩
    obtain ⟨value, returned⟩ := stable.returned handle ⟨pair, status⟩ stored
    cases returned
    by_cases order : pair.2.entry.identity < pair.1.entry.identity
    · apply swapped (pair.2, pair.1)
      simpa only [order, ↓reduceIte] using admitted handle pair value stored
    · simpa only [order, ↓reduceIte] using admitted handle pair value stored
