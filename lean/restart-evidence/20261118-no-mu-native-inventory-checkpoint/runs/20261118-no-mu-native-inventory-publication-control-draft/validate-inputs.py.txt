SHA = lambda data: hashlib.sha256(data).hexdigest()
read = lambda path: json.loads(path.read_text())
bound = set()
OVERRIDES = globals().get('OVERRIDES',{})

def records(value):
    if isinstance(value,list):
        return [(e.get('file',e.get('path')),e) for e in value]
    for key in ['files','pins','immutable_dependencies']:
        if key in value:
            return records(value[key])
    return value.items()

def verify(path,record):
    data = path.read_bytes()
    assert SHA(data) == (record['sha256'] if isinstance(record,dict) else record),str(path)
    if isinstance(record,dict) and 'bytes' in record:
        assert len(data) == record['bytes'],str(path)
    bound.add(path)
    return data

def packet(path,digest=None):
    manifest = read(path/'MANIFEST.json') if digest is None else json.loads(verify(path/'MANIFEST.json',digest))
    for name,record in records(manifest):
        verify(path/name,record)
    bound.add(path/'MANIFEST.json')

def pins(path,digest=None,count=None):
    value = json.loads(verify(path,digest)) if digest else read(path)
    value = dict(records(value))
    if count is not None:
        assert len(value) == count
    for name,record in value.items():
        verify(ROOT/OVERRIDES.get(name,name),record)
    return value
