590: /-- Retain ordered edge receipts, continuation outputs and the exact stopping context. -/
591: inductive Result (Output : Type u) (Reason : Type v) (width : Nat) where
592:   /-- Every edge and invoked continuation completed. -/
593:   | complete (processed : List (Receipt width)) (outputs : List Output)
594:       (caller : Caller width) (queries : Queries)
595:   /-- Retain the finished prefix and the current edge or continuation stop. -/
596:   | stopped (processed : List (Receipt width)) (outputs : List Output) (current : Stop Reason width)

598: /-- Valid rejections retain their actual caller without invoking a fragment continuation. -/
599: def advance : Receipt width → Option (Caller width × Queries)
600:   | .complete _ caller queries _ => some (caller, queries)
601:   | .rejected caller queries _ => some (caller, queries)
602:   | .stopped _ (.ok (some []), caller, queries, _outer, _waiting) (some _) => some (caller, queries)
603:   | _ => none

605: /-- Read the actual child endpoint or retain the entry endpoint of an unsupported pure prefix. -/
606: def endpoint (receipt : Receipt width) (caller : Caller width) (queries : Queries) : Caller width × Queries :=
607:   match receipt with
608:   | .complete _ reached saved _ => (reached, saved)
609:   | .rejected reached saved _ => (reached, saved)
610:   | .stopped _ (_, reached, saved, _, _) _ => (reached, saved)
611:   | .finalizeStopped _ reached saved _ => (reached, saved)
612:   | .unhandled => (caller, queries)
