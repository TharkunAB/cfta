32: def erasePrefix {Typing : Type} : Result width Typing → Result width Unit
33:   | .stopped read checked => .stopped read checked
34:   | .parked checked candidate raw =>
35:       .parked checked candidate (raw.map fun rows => rows.map fun
36:         | .error fault => .error fault
37:         | .ok ((metadata, next), after) => .ok ((eraseMetadata metadata, next), after))
38:   | .narrowed checked prepared caller queries action candidate right ordinary =>
39:       .narrowed checked prepared caller queries action candidate
40:         (right.map fun rows => rows.map fun
41:           | .error fault => .error fault
42:           | .ok ((flag, metadata), after) => .ok ((flag, eraseMetadata metadata), after)) ordinary
