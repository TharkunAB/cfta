41: /-- Invoke each completed child's tuple continuation before the next ordinary edge. -/
42: def consumeSelected
43:     (done : Fragment → Caller width → Queries → Cached → Result Output Reason width)
44:     (caller : Caller width) (queries : Queries) (receipt : Receipt width) : Result Output Reason width :=
45:   match receipt with
46:   | .complete fragment reached saved after =>
47:       match done fragment reached saved after with
48:       | .complete processed outputs finalCaller finalQueries =>
49:           .complete (receipt :: processed) outputs finalCaller finalQueries
50:       | .stopped processed outputs current => .stopped (receipt :: processed) outputs current
51:   | other =>
52:       match advance other with
53:       | none => .stopped [] [] (.edge (some other) (endpoint other caller queries))
54:       | some (reached, saved) => .complete [other] [] reached saved
