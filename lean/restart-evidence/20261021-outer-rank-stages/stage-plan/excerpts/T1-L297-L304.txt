  298 theorem lex_def {r : α → α → Prop} {s : β → β → Prop} {p q : α × β} :
  299     Prod.Lex r s p q ↔ r p.1 q.1 ∨ p.1 = q.1 ∧ s p.2 q.2 :=
  300   ⟨fun h => by cases h <;> simp [*], fun h =>
  301     match p, q, h with
  302     | _, _, Or.inl h => Lex.left _ _ h
  303     | (_, _), (_, _), Or.inr ⟨e, h⟩ => by subst e; exact Lex.right _ h⟩
