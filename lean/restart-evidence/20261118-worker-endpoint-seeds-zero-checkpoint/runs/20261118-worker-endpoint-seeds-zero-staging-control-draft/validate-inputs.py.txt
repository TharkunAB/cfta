SHA = lambda data: hashlib.sha256(data).hexdigest()

bound = set()

def verify(path, record):
    data = path.read_bytes()
    assert SHA(data) == (record['sha256'] if isinstance(record,dict) else record),str(path)
    if isinstance(record,dict) and 'bytes' in record:
        assert len(data) == record['bytes'],str(path)
    bound.add(path)
    return data


def records(value):
    if isinstance(value,list):
        return [(item.get('file',item.get('path')),item) for item in value]
    if 'files' in value:
        return records(value['files'])
    if 'pins' in value:
        return records(value['pins'])
    return value.items()


def historical(name, record):
    digest = record['sha256'] if isinstance(record,dict) else record
    for path in [ROOT/name,ROOT/OVERRIDES.get(name,name)]:
        if path.is_file() and SHA(path.read_bytes()) == digest:
            return verify(path,record)
    raise AssertionError(('Unmapped historical pin',name))


def packet(path, digest=None):
    manifest = json.loads(verify(path/'MANIFEST.json',digest or SHA((path/'MANIFEST.json').read_bytes())))
    for name,record in records(manifest['files']):
        verify(path/name,record)
    return manifest


def pin_map(path, digest=None, count=None):
    values = json.loads(verify(path,digest or SHA(path.read_bytes())))
    if count is not None:
        assert len(values) == count
    for name,record in values.items():
        historical(name,record)
    return values


def clean_check(item, reports, declarations):
    assert item['exit'] == 0 and item['total_axiom_reports'] == reports
    assert not item['diagnostics'] and not item['sorryAx'] and not item['nonstandard_axioms']
    assert item['declaration_checks'] == declarations
    assert item['axiom_groups'] + item['axiom_free_reports'] == reports


def private_check(work, verified_name, actual_name, declarations):
    verified,actual = read(work/verified_name),read(work/actual_name)
    assert verified['actual_terminal'] and actual['actual_terminal']
    assert verified['actual_terminal_exit'] == actual['actual_terminal_exit'] == 0
    assert verified['actual_session'] == actual['actual_session']
    assert verified['source_commit'] == SOURCE and not verified['normal_Source_changed']
    assert verified['credited_public_reports'] == 0 and verified['credited_private_reports'] == 1
    assert verified['proof_credit'] and verified['source_pins_unchanged']
    assert verified['definition_axiom_credit'] == 0
    runtime = BASE/verified['runtime']
    checked = json.loads(verify(runtime/'checked.json',verified['checked_sha256']))
    assert checked == json.loads(verify(runtime/'results.json',verified['results_sha256']))
    assert checked['checks'] == verified['checks'] and checked['limits'] == LIMITS
    assert len(checked['checks']) == 1
    clean_check(checked['checks'][0],1,declarations)
    pin_map(runtime/'source-pins.json',verified['source_pins_sha256'],verified['pins_independently_reverified'])
    control = BASE/verified['selected_control']
    assert verified['selected_control'] == actual['selected_control']
    assert verified['selected_control_manifest_sha256'] == actual['selected_control_manifest_sha256']
    packet(control,verified['selected_control_manifest_sha256'])
    for path in [work/verified_name,work/actual_name,runtime/(checked['checks'][0]['name']+'.txt')]:
        bound.add(path)
    return verified,actual,runtime,checked['checks'][0]
